  <script>
    (() => {
      const byId = (id) => document.getElementById(id);
      const elements = Object.fromEntries([
        'rear-wheel', 'front-wheel', 'crank', 'rider', 'near-leg-outline', 'near-leg-line',
        'far-leg-outline', 'far-leg-line', 'near-foot', 'far-foot', 'clouds', 'waves',
        'moving-road', 'front-plants', 'back-plants', 'scarf-tail', 'eye', 'closed-eye',
        'wind', 'far-birds', 'wing', 'wing-detail', 'bike-shadow'
      ].map((id) => [id, byId(id)]));
      const playButton = byId('play-button');
      const speedButtons = Array.from(document.querySelectorAll('[data-speed]'));
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      let playing = !reducedMotion.matches;
      let speed = 1;
      let elapsed = 0;
      let previousTime = null;
      let frameId = null;
      const setTransform = (id, transform) => elements[id].setAttribute('transform', transform);

      function drawLeg(side, angle, bob) {
        const footX = 557 + Math.cos(angle) * 31;
        const footY = 527 + Math.sin(angle) * 31;
        const hipX = side === 'near' ? 536 : 529;
        const hipY = 399 + bob;
        const ankleY = footY - 12;
        const dx = footX - hipX;
        const dy = ankleY - hipY;
        const distance = Math.hypot(dx, dy);
        const bend = Math.sqrt(84 * 84 - distance * distance / 4);
        const kneeX = (hipX + footX) / 2 + dy / distance * bend;
        const kneeY = (hipY + ankleY) / 2 - dx / distance * bend;
        const path = `M${hipX} ${hipY} Q${kneeX + 3} ${kneeY - 4} ${kneeX} ${kneeY} L${footX} ${ankleY}`;
        elements[`${side}-leg-outline`].setAttribute('d', path);
        elements[`${side}-leg-line`].setAttribute('d', path);
        setTransform(`${side}-foot`, `translate(${footX} ${footY})`);
      }

      function render(time) {
        const phase = time * 2.65;
        const degrees = phase * 180 / Math.PI;
        const bob = Math.sin(phase * 2) * 2;
        setTransform('rear-wheel', `rotate(${degrees * 1.85})`);
        setTransform('front-wheel', `rotate(${degrees * 1.85})`);
        setTransform('crank', `rotate(${degrees} 557 527)`);
        setTransform('rider', `translate(0 ${bob})`);
        drawLeg('near', phase, bob);
        drawLeg('far', phase + Math.PI, bob);
        setTransform('clouds', `translate(${-((time * 8) % 1220)} 0)`);
        setTransform('waves', `translate(${-((time * 17) % 480)} 0)`);
        setTransform('moving-road', `translate(${-((time * 90 * 2.65 * 1.85) % 440)} 0)`);
        setTransform('back-plants', `translate(${-((time * 75) % 1260)} 0)`);
        setTransform('front-plants', `translate(${-((time * 190) % 1260)} 0)`);
        setTransform('wind', `translate(${Math.sin(time * 2) * 8} 0)`);
        setTransform('far-birds', `translate(${Math.sin(time * .3) * 9} ${Math.sin(time * .8) * 3})`);
        elements['bike-shadow'].setAttribute('rx', 231 + bob * 1.4);
        const flutter = Math.sin(time * 7) * 6;
        elements['scarf-tail'].setAttribute('d', `M582 281C554 ${265 + flutter} 542 ${291 - flutter} 514 ${275 + flutter}L488 ${278 + flutter}l15 12c29 ${9 - flutter} 51-7 80-1Z`);
        elements.wing.setAttribute('d', `M501 343c29-8 47 6 66 20C591 380 626 ${355 - bob} 661 ${353 - bob}q20-2 20 7c-1 5-10 10-18 10C633 ${380 - bob} 611 401 579 399c-31-2-63-17-72-31`);
        const blinking = time % 5.7 > 5.5;
        elements.eye.setAttribute('opacity', blinking ? '0' : '1');
        elements['closed-eye'].setAttribute('opacity', blinking ? '1' : '0');
      }

      function tick(timestamp) {
        if (!playing || document.hidden) { frameId = null; return; }
        if (previousTime !== null) elapsed += Math.min((timestamp - previousTime) / 1000, .05) * speed;
        previousTime = timestamp;
        render(elapsed);
        frameId = requestAnimationFrame(tick);
      }

      function schedule() {
        previousTime = null;
        if (frameId !== null) cancelAnimationFrame(frameId);
        frameId = playing && !document.hidden ? requestAnimationFrame(tick) : null;
      }

      function updatePlayback() {
        playButton.classList.toggle('is-paused', !playing);
        playButton.setAttribute('aria-label', playing ? '暂停动画' : '播放动画');
        playButton.setAttribute('aria-pressed', String(playing));
        byId('status').textContent = playing ? '海边兜风中' : '停下来，看看海';
        byId('status-sub').textContent = playing ? '风很轻，心情刚刚好。' : '好风景，值得多停一会儿。';
        schedule();
      }

      playButton.addEventListener('click', () => { playing = !playing; updatePlayback(); });
      speedButtons.forEach((button) => button.addEventListener('click', () => {
        speed = Number(button.dataset.speed);
        speedButtons.forEach((option) => option.setAttribute('aria-pressed', String(option === button)));
      }));
      reducedMotion.addEventListener('change', (event) => {
        if (event.matches) { playing = false; updatePlayback(); }
      });
      document.addEventListener('visibilitychange', schedule);
      render(0);
      updatePlayback();
    })();
  </script>
