<script>
(function () {
  const $ = id => document.getElementById(id);

  const BB = { x: 370, y: 372 };
  const HIP = { x: 345, y: 262 };
  const CRANK_R = 30, WHEEL_R = 52, THIGH = 75, SHIN = 75;
  const SPEED = 130;

  const rear = $('rearWheel'), front = $('frontWheel');
  const roadLine = $('roadLine'), chain = $('chain');
  const rider = $('rider'), head = $('head'), pouch = $('pouch');
  const layers = [
    [$('clouds'), 0.08],
    [$('hillsFar'), 0.2],
    [$('hillsNear'), 0.35],
    [$('trees'), 0.6]
  ];

  function solveKnee(hip, ankle) {
    const dx = ankle.x - hip.x, dy = ankle.y - hip.y;
    const d = Math.min(Math.max(Math.hypot(dx, dy), Math.abs(THIGH - SHIN) + 0.01), THIGH + SHIN - 0.01);
    const base = Math.atan2(dy, dx);
    const a = Math.acos((THIGH * THIGH + d * d - SHIN * SHIN) / (2 * THIGH * d));
    const k1 = { x: hip.x + THIGH * Math.cos(base - a), y: hip.y + THIGH * Math.sin(base - a) };
    const k2 = { x: hip.x + THIGH * Math.cos(base + a), y: hip.y + THIGH * Math.sin(base + a) };
    return k1.x > k2.x ? k1 : k2;
  }

  function updateLeg(prefix, ang, hipY) {
    const pedal = { x: BB.x + CRANK_R * Math.cos(ang), y: BB.y + CRANK_R * Math.sin(ang) };
    const ankle = { x: pedal.x - 4, y: pedal.y - 8 };
    const hip = { x: HIP.x, y: hipY };
    const knee = solveKnee(hip, ankle);
    $(prefix + 'Thigh').setAttribute('d', `M${hip.x} ${hip.y} L${knee.x} ${knee.y}`);
    $(prefix + 'Shin').setAttribute('d', `M${knee.x} ${knee.y} L${ankle.x} ${ankle.y}`);
    $(prefix + 'Foot').setAttribute('transform', `translate(${pedal.x} ${pedal.y})`);
    $(prefix + 'Crank').setAttribute('d', `M${BB.x} ${BB.y} L${pedal.x} ${pedal.y}`);
    $(prefix + 'Pedal').setAttribute('transform', `translate(${pedal.x} ${pedal.y})`);
  }

  let dist = 0, last = null;

  function frame(t) {
    if (last === null) last = t;
    const dt = Math.min((t - last) / 1000, 0.05);
    last = t;
    dist += SPEED * dt;

    const wheelAng = dist / WHEEL_R;
    const crankAng = wheelAng / 2;
    const deg = wheelAng * 180 / Math.PI;
    rear.setAttribute('transform', `translate(270 378) rotate(${deg})`);
    front.setAttribute('transform', `translate(520 378) rotate(${deg})`);

    for (const [el, f] of layers) {
      el.setAttribute('transform', `translate(${-((dist * f) % 900)} 0)`);
    }
    roadLine.setAttribute('stroke-dashoffset', dist);
    chain.setAttribute('stroke-dashoffset', -crankAng * 20);

    const bob = Math.sin(crankAng * 2) * 1.8;
    rider.setAttribute('transform', `translate(0 ${bob})`);
    head.setAttribute('transform', `rotate(${Math.sin(crankAng * 2 + 0.6) * 2.5} 452 134)`);
    const pouchScale = 1 + Math.sin(crankAng * 2 + 1.2) * 0.06;
    pouch.setAttribute('transform', `translate(474 128) scale(1 ${pouchScale}) translate(-474 -128)`);

    updateLeg('near', crankAng, HIP.y + bob);
    updateLeg('far', crankAng + Math.PI, HIP.y + bob);

    requestAnimationFrame(frame);
  }

  requestAnimationFrame(frame);
})();
</script>
