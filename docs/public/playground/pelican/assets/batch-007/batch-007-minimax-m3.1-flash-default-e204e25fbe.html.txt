<script>
(function () {
  var svgNS = 'http://www.w3.org/2000/svg';
  var $ = function (s) { return document.querySelector(s); };

  var scene   = $('#scene');
  var crank   = $('#crank');
  var pelican = $('#pelican');
  var pouch   = $('#pouch');
  var legNear = $('#legNear'), footNear = $('#footNear');
  var legFar  = $('#legFar'),  footFar  = $('#footFar');
  var pedalNear = $('#pedalNear'), pedalFar = $('#pedalFar');
  var scenery = $('#scenery');

  var BB = { x: 430, y: 364 };   // 中轴
  var R  = 30;                   // 曲柄半径
  var THIGH = 95, SHIN = 92;     // 大腿 / 小腿长度
  var HIP = { x: 438, y: 214 };  // 髋关节
  var CRANK_PERIOD = 1.6;        // 秒 / 圈

  /* ---------- 辐条 ---------- */
  function buildSpokes(id, cx, cy, r) {
    var g = $(id), s = '';
    for (var i = 0; i < 12; i++) {
      var a = i * Math.PI / 6;
      s += '<line x1="' + cx + '" y1="' + cy + '" x2="' + (cx + r * Math.cos(a)).toFixed(2) +
           '" y2="' + (cy + r * Math.sin(a)).toFixed(2) + '" stroke="#b7c2cd" stroke-width="1.6"/>';
    }
    g.innerHTML = s;
  }
  buildSpokes('#spokesRear', 300, 366, 51);
  buildSpokes('#spokesFront', 640, 366, 51);

  /* ---------- 远景树丛 ---------- */
  function treeRow(offset) {
    var s = '', spots = [[40,1],[150,.85],[262,1.05],[392,.9],[500,1],[628,.82],[742,1.02],[858,.9]];
    for (var i = 0; i < spots.length; i++) {
      var x = spots[i][0] + offset, k = spots[i][1];
      var h = 26 * k, r = 17 * k;
      s += '<g transform="translate(' + x + ' 404)">' +
           '<path d="M0,0 L0,' + (-h) + '" stroke="#8a6a44" stroke-width="' + (5 * k).toFixed(1) + '"/>' +
           '<circle cx="0" cy="' + (-h - r * .6).toFixed(1) + '" r="' + r.toFixed(1) + '" fill="#4f9e63"/>' +
           '<circle cx="' + (-r * .6).toFixed(1) + '" cy="' + (-h - r * .2).toFixed(1) + '" r="' + (r * .72).toFixed(1) + '" fill="#5cb073"/>' +
           '<circle cx="' + (r * .62).toFixed(1) + '" cy="' + (-h - r * .18).toFixed(1) + '" r="' + (r * .68).toFixed(1) + '" fill="#5cb073"/>' +
           '</g>';
    }
    return s;
  }
  scenery.innerHTML = treeRow(0) + treeRow(900);

  /* ---------- 两骨 IK ---------- */
  function ik(hx, hy, fx, fy, bend) {
    var dx = fx - hx, dy = fy - hy;
    var d = Math.hypot(dx, dy);
    var max = (THIGH + SHIN) * 0.999;
    if (d > max) { var k = max / d; fx = hx + dx * k; fy = hy + dy * k; dx *= k; dy *= k; d = max; }
    var a = Math.acos(Math.max(-1, Math.min(1, (THIGH * THIGH + d * d - SHIN * SHIN) / (2 * THIGH * d))));
    var ang = Math.atan2(dy, dx) + a * bend;
    return { kx: hx + Math.cos(ang) * THIGH, ky: hy + Math.sin(ang) * THIGH, fx: fx, fy: fy };
  }

  /* ---------- 每帧更新 ---------- */
  function update(phi) {
    var deg = phi * 180 / Math.PI;
    crank.setAttribute('transform', 'rotate(' + deg.toFixed(3) + ' 430 364)');

    var bob = Math.sin(phi * 2) * 2.2;
    pelican.setAttribute('transform', 'translate(0 ' + bob.toFixed(2) + ')');

    var nx = BB.x + R * Math.cos(phi), ny = BB.y + R * Math.sin(phi);
    var fx = BB.x - R * Math.cos(phi), fy = BB.y - R * Math.sin(phi);

    pedalNear.setAttribute('transform', 'translate(' + nx.toFixed(2) + ' ' + ny.toFixed(2) + ')');
    pedalFar.setAttribute('transform', 'translate(' + fx.toFixed(2) + ' ' + fy.toFixed(2) + ')');

    var hy = HIP.y + bob;
    var n = ik(HIP.x, hy, nx, ny - 8, -1);
    var f = ik(HIP.x - 9, hy + 2, fx, fy - 8, -1);

    legNear.setAttribute('d', 'M ' + HIP.x + ' ' + hy.toFixed(2) + ' L ' + n.kx.toFixed(2) + ' ' + n.ky.toFixed(2) + ' L ' + n.fx.toFixed(2) + ' ' + n.fy.toFixed(2));
    footNear.setAttribute('d', 'M ' + n.fx.toFixed(2) + ' ' + n.fy.toFixed(2) + ' L ' + (n.fx + 14).toFixed(2) + ' ' + (n.fy - 5).toFixed(2));
    legFar.setAttribute('d', 'M ' + (HIP.x - 9) + ' ' + (hy + 2).toFixed(2) + ' L ' + f.kx.toFixed(2) + ' ' + f.ky.toFixed(2) + ' L ' + f.fx.toFixed(2) + ' ' + f.fy.toFixed(2));
    footFar.setAttribute('d', 'M ' + f.fx.toFixed(2) + ' ' + f.fy.toFixed(2) + ' L ' + (f.fx + 12).toFixed(2) + ' ' + (f.fy - 5).toFixed(2));

    var p = -2.2 + 2.6 * Math.sin(phi * 2 + 0.7);
    pouch.setAttribute('transform', 'rotate(' + p.toFixed(2) + ')');
  }

  /* ---------- 时间循环 ---------- */
  var elapsed = 0, last = null, running = true;
  function frame(now) {
    if (last !== null && running) elapsed += (now - last) / 1000;
    last = now;
    if (running) update((elapsed / CRANK_PERIOD) * Math.PI * 2);
    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);

  document.addEventListener('keydown', function (e) {
    if (e.code === 'Space') { e.preventDefault(); running = !running; }
  });
})();
</script>
