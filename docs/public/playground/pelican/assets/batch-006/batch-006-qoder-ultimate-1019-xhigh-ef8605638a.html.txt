  <script>
    const app = document.getElementById('app');
    const scene = document.getElementById('scene');
    const toggle = document.getElementById('play-toggle');
    const playTitle = document.getElementById('play-title');
    const sceneStatus = document.getElementById('scene-status');
    const pelican = document.getElementById('pelican');
    const cranks = document.getElementById('cranks');
    const wheels = [document.getElementById('rear-wheel-spin'), document.getElementById('front-wheel-spin')];
    const speedButtons = [...document.querySelectorAll('.speed-button')];
    const legs = ['far', 'near'].map(side => ({
      path: document.getElementById(`${side}-leg-path`),
      foot: document.getElementById(`${side}-foot`),
      pedal: document.getElementById(`${side}-pedal`)
    }));
    const motionPreference = window.matchMedia('(prefers-reduced-motion: reduce)');
    const mobileLayout = window.matchMedia('(max-width: 700px)');
    let running = !motionPreference.matches;
    let speed = 1;
    let phase = 0;
    let wheelAngle = 0;
    let lastTime = null;
    let animationFrame = null;

    function drawLeg(leg, hipX, hipY, angle, upperLength, lowerLength) {
      const x = 587 + Math.cos(angle) * 24;
      const y = 474 + Math.sin(angle) * 24;
      const dx = x - hipX;
      const dy = y - hipY;
      const distance = Math.hypot(dx, dy);
      const along = (upperLength ** 2 - lowerLength ** 2 + distance ** 2) / (2 * distance);
      const height = Math.sqrt(Math.max(0, upperLength ** 2 - along ** 2));
      const kneeX = hipX + along * dx / distance + height * dy / distance;
      const kneeY = hipY + along * dy / distance - height * dx / distance;
      leg.path.setAttribute('d', `M${hipX} ${hipY} L${kneeX} ${kneeY} L${x} ${y - 5}`);
      leg.foot.setAttribute('d', `M${x - 10} ${y - 4} Q${x} ${y - 1} ${x + 13} ${y - 4}`);
      leg.pedal.setAttribute('d', `M${x - 11} ${y + 2} H${x + 11}`);
    }

    function render() {
      const bob = Math.sin(phase * 2) * 1.7;
      pelican.setAttribute('transform', `translate(0 ${bob})`);
      drawLeg(legs[0], 566, 357 + bob, phase + Math.PI, 86, 80);
      drawLeg(legs[1], 580, 360 + bob, phase, 83, 78);
      cranks.setAttribute('transform', `rotate(${phase * 180 / Math.PI} 587 474)`);
      wheels.forEach(wheel => wheel.setAttribute('transform', `rotate(${wheelAngle})`));
    }

    function animate(time) {
      if (!running) return;
      if (lastTime !== null) {
        const delta = Math.min(time - lastTime, 64);
        const advance = delta / 1250 * Math.PI * 2 * speed;
        phase = (phase + advance) % (Math.PI * 2);
        wheelAngle = (wheelAngle + advance * 180 / Math.PI * 1.65) % 360;
      }
      lastTime = time;
      render();
      animationFrame = requestAnimationFrame(animate);
    }

    function setRunning(value) {
      running = value;
      app.dataset.paused = String(!running);
      toggle.setAttribute('aria-label', running ? '暂停动画' : '继续骑行');
      toggle.setAttribute('aria-pressed', String(!running));
      playTitle.textContent = running ? '正在兜风' : '歇一会儿，看看海';
      sceneStatus.textContent = running ? 'ON A LITTLE RIDE' : 'A LITTLE BREAK';
      cancelAnimationFrame(animationFrame);
      lastTime = null;
      if (running) animationFrame = requestAnimationFrame(animate);
    }

    toggle.addEventListener('click', () => setRunning(!running));
    speedButtons.forEach(button => button.addEventListener('click', () => {
      speed = Number(button.dataset.speed);
      app.style.setProperty('--cycle-duration', `${1250 / speed}ms`);
      app.style.setProperty('--scene-duration', `${9000 / speed}ms`);
      speedButtons.forEach(option => option.setAttribute('aria-pressed', String(option === button)));
    }));
    document.addEventListener('keydown', event => {
      if (event.code === 'Space' && !event.repeat && !event.target.closest('button, input, textarea, select, a, [contenteditable="true"]')) {
        event.preventDefault();
        setRunning(!running);
      }
    });
    document.addEventListener('visibilitychange', () => { lastTime = null; });
    motionPreference.addEventListener('change', event => setRunning(!event.matches));
    function setViewBox() {
      scene.setAttribute('viewBox', mobileLayout.matches ? '300 30 620 660' : '0 0 1200 620');
    }
    mobileLayout.addEventListener('change', setViewBox);
    setViewBox();
    render();
    setRunning(running);
  </script>
