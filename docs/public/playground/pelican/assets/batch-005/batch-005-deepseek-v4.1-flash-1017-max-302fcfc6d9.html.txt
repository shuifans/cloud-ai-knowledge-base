<script>
(() => {
  const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const NS = 'http://www.w3.org/2000/svg';
  const $ = (id) => document.getElementById(id);

  const rearWheel = { x: 436, y: 524 };
  const frontWheel = { x: 812, y: 524 };
  const bottomBracket = { x: 634, y: 478 };
  const crankRadius = 42;
  const nearHip = { x: 590, y: 333 };
  const farHip = { x: 607, y: 330 };
  const nearThigh = 96;
  const nearShin = 103;
  const farThigh = 91;
  const farShin = 98;

  const spokes = [
    { group: $('rearSpokes'), center: rearWheel },
    { group: $('frontSpokes'), center: frontWheel }
  ];

  spokes.forEach(({ group, center }) => {
    for (let i = 0; i < 16; i++) {
      const angle = (i / 16) * Math.PI * 2;
      const x1 = center.x + Math.cos(angle) * 10;
      const y1 = center.y + Math.sin(angle) * 10;
      const x2 = center.x + Math.cos(angle) * 82;
      const y2 = center.y + Math.sin(angle) * 82;

      const under = document.createElementNS(NS, 'line');
      under.setAttribute('x1', x1);
      under.setAttribute('y1', y1);
      under.setAttribute('x2', x2);
      under.setAttribute('y2', y2);
      under.setAttribute('stroke', '#173a33');
      under.setAttribute('stroke-width', '3.4');
      under.setAttribute('stroke-linecap', 'round');
      under.setAttribute('opacity', '0.16');
      group.appendChild(under);

      const spoke = document.createElementNS(NS, 'line');
      spoke.setAttribute('x1', x1);
      spoke.setAttribute('y1', y1);
      spoke.setAttribute('x2', x2);
      spoke.setAttribute('y2', y2);
      spoke.setAttribute('stroke', '#fff0d1');
      spoke.setAttribute('stroke-width', '1.8');
      spoke.setAttribute('stroke-linecap', 'round');
      spoke.setAttribute('opacity', '0.9');
      group.appendChild(spoke);
    }

    const reflectorAngle = -0.6;
    const reflector = document.createElementNS(NS, 'circle');
    reflector.setAttribute('cx', center.x + Math.cos(reflectorAngle) * 62);
    reflector.setAttribute('cy', center.y + Math.sin(reflectorAngle) * 62);
    reflector.setAttribute('r', '4.6');
    reflector.setAttribute('fill', '#e45f48');
    reflector.setAttribute('stroke', '#173a33');
    reflector.setAttribute('stroke-width', '1.6');
    group.appendChild(reflector);
  });

  const elements = {
    rig: $('rig'),
    headBob: $('headBob'),
    scarfTailTop: $('scarfTailTop'),
    scarfTailBottom: $('scarfTailBottom'),
    eyeBall: $('eyeBall'),
    eyeShine: $('eyeShine'),
    nearLegOutline: $('nearLegOutline'),
    nearLegColor: $('nearLegColor'),
    farLegOutline: $('farLegOutline'),
    farLegColor: $('farLegColor'),
    nearFoot: $('nearFoot'),
    farFoot: $('farFoot'),
    nearCrankArm: $('nearCrankArm'),
    farCrankArm: $('farCrankArm'),
    nearPedal: $('nearPedal'),
    farPedal: $('farPedal'),
    chainring: $('chainring'),
    chainPath: $('chainPath'),
    trailScroller: $('trailScroller'),
    vergeScroller: $('vergeScroller'),
    fgScroller: $('fgScroller')
  };

  const clouds = Array.from(document.querySelectorAll('.cloud')).map((node) => ({
    node,
    x: Number(node.dataset.x),
    y: Number(node.dataset.y),
    scale: Number(node.dataset.scale),
    speed: Number(node.dataset.speed)
  }));

  let crankAngle = 0.68;
  let wheelAngle = 14;
  let chainPhase = 2;
  let trailOffset = 0;
  let vergeOffset = 0;
  let foregroundOffset = 0;
  let blinkClock = 0;
  let lastTime = 0;
  let running = false;

  function solveKnee(hip, foot, upper, lower) {
    const dx = foot.x - hip.x;
    const dy = foot.y - hip.y;
    const distance = Math.max(1, Math.min(Math.hypot(dx, dy), upper + lower - 0.2));
    const baseDistance = (upper * upper - lower * lower + distance * distance) / (2 * distance);
    const height = Math.sqrt(Math.max(0, upper * upper - baseDistance * baseDistance));
    const ux = dx / distance;
    const uy = dy / distance;
    const baseX = hip.x + ux * baseDistance;
    const baseY = hip.y + uy * baseDistance;
    let px = -uy;
    let py = ux;
    if (px < 0) {
      px = -px;
      py = -py;
    }
    return {
      x: baseX + px * height,
      y: baseY + py * height
    };
  }

  function legPath(hip, knee, foot) {
    return `M ${hip.x.toFixed(2)} ${hip.y.toFixed(2)} L ${knee.x.toFixed(2)} ${knee.y.toFixed(2)} L ${foot.x.toFixed(2)} ${foot.y.toFixed(2)}`;
  }

  function pedalPosition(angle) {
    return {
      x: bottomBracket.x + Math.cos(angle) * crankRadius,
      y: bottomBracket.y + Math.sin(angle) * crankRadius
    };
  }

  function placePedal(arm, pedal, foot, hip, upper, lower) {
    const position = pedalPosition(arm === elements.nearCrankArm ? crankAngle : crankAngle + Math.PI);
    arm.setAttribute('x1', bottomBracket.x);
    arm.setAttribute('y1', bottomBracket.y);
    arm.setAttribute('x2', position.x.toFixed(2));
    arm.setAttribute('y2', position.y.toFixed(2));
    pedal.setAttribute('transform', `translate(${position.x.toFixed(2)} ${position.y.toFixed(2)})`);

    const footPoint = { x: position.x + 1.5, y: position.y - 5 };
    const knee = solveKnee(hip, footPoint, upper, lower);
    foot.setAttribute('transform', `translate(${footPoint.x.toFixed(2)} ${footPoint.y.toFixed(2)})`);
    return legPath(hip, knee, footPoint);
  }

  function updateScene(time, dt) {
    if (dt > 0) {
      crankAngle = (crankAngle + dt * 2.55) % (Math.PI * 2);
      wheelAngle = (wheelAngle + dt * 4.2) % 360;
      chainPhase = (chainPhase + dt * 30.6) % 7;
      trailOffset = (trailOffset + dt * 300) % 1200;
      vergeOffset = (vergeOffset + dt * 74) % 1200;
      foregroundOffset = (foregroundOffset + dt * 440) % 1200;
      blinkClock = (blinkClock + dt) % 5.2;

      clouds.forEach((cloud) => {
        cloud.x -= cloud.speed * dt;
        if (cloud.x < -190) cloud.x += 1460;
      });
    }

    const bob = Math.sin(time * 4.6) * 1.35 + Math.sin(time * 2.15 + 0.9) * 1.1;
    const lean = Math.sin(time * 2.15 + 0.9) * 0.12;
    elements.rig.setAttribute('transform', `translate(0 ${bob.toFixed(2)}) rotate(${lean.toFixed(3)} 620 530)`);

    const headNod = Math.sin(time * 2.15 + 1.5) * 0.8 + Math.sin(time * 4.6 + 0.3) * 0.22;
    const headLift = Math.sin(time * 4.6 + 0.1) * 0.7;
    elements.headBob.setAttribute('transform', `translate(0 ${headLift.toFixed(2)}) rotate(${headNod.toFixed(3)} 682 251)`);

    const scarfTop = -1.4 + Math.sin(time * 4.7 + 0.4) * 4.6 + Math.sin(time * 8.2) * 0.9;
    const scarfBottom = 1.6 + Math.sin(time * 5.5 + 2.1) * 4.2;
    elements.scarfTailTop.setAttribute('transform', `rotate(${scarfTop.toFixed(2)} 656 241)`);
    elements.scarfTailBottom.setAttribute('transform', `rotate(${scarfBottom.toFixed(2)} 656 243)`);

    const blinkProgress = blinkClock > 4.94 ? Math.sin(((blinkClock - 4.94) / 0.26) * Math.PI) : 0;
    const eyeHeight = Math.max(0.7, 8.4 * (1 - blinkProgress));
    elements.eyeBall.setAttribute('ry', eyeHeight.toFixed(2));
    elements.eyeShine.setAttribute('opacity', blinkProgress > 0.24 ? '0' : '1');

    elements.chainring.setAttribute('transform', `rotate(${((crankAngle * 180) / Math.PI).toFixed(2)} 634 478)`);
    elements.chainPath.setAttribute('stroke-dashoffset', (-chainPhase).toFixed(2));
    $('rearSpokes').setAttribute('transform', `rotate(${wheelAngle.toFixed(2)} ${rearWheel.x} ${rearWheel.y})`);
    $('frontSpokes').setAttribute('transform', `rotate(${wheelAngle.toFixed(2)} ${frontWheel.x} ${frontWheel.y})`);

    const nearPath = placePedal(elements.nearCrankArm, elements.nearPedal, elements.nearFoot, nearHip, nearThigh, nearShin);
    const farPath = placePedal(elements.farCrankArm, elements.farPedal, elements.farFoot, farHip, farThigh, farShin);
    elements.nearLegOutline.setAttribute('d', nearPath);
    elements.nearLegColor.setAttribute('d', nearPath);
    elements.farLegOutline.setAttribute('d', farPath);
    elements.farLegColor.setAttribute('d', farPath);

    elements.trailScroller.setAttribute('transform', `translate(${(-trailOffset).toFixed(2)} 0)`);
    elements.vergeScroller.setAttribute('transform', `translate(${(-vergeOffset).toFixed(2)} 0)`);
    elements.fgScroller.setAttribute('transform', `translate(${(-foregroundOffset).toFixed(2)} 0)`);

    clouds.forEach((cloud) => {
      cloud.node.setAttribute('transform', `translate(${cloud.x.toFixed(2)} ${cloud.y}) scale(${cloud.scale})`);
    });
  }

  function frame(now) {
    if (!running) return;
    if (!lastTime) lastTime = now;
    const dt = Math.min((now - lastTime) / 1000, 0.05);
    lastTime = now;
    updateScene(now / 1000, dt);
    requestAnimationFrame(frame);
  }

  updateScene(0, 0);

  if (reducedMotion) {
    updateScene(1.8, 0);
  } else {
    running = true;
    requestAnimationFrame((now) => {
      lastTime = now;
      updateScene(now / 1000, 0);
      requestAnimationFrame(frame);
    });
  }
})();
