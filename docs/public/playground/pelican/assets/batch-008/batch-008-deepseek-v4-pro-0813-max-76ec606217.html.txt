<script>
(function () {
  var PERIOD = 0.9, BX = 390, BY = 385, R = 28, L1 = 65, L2 = 65;
  var HIP_N = { x: 330, y: 301 }, HIP_F = { x: 326, y: 299 }, BC = { x: 330, y: 275 };
  var $ = function (id) { return document.getElementById(id); };
  var wheelR = $('wheelRSpokes'), wheelF = $('wheelFSpokes');
  var crankF = $('crankFar'), crankN = $('crankNear');
  var bodyG = $('bodyG'), pouch = $('pouch'), scarf = $('scarf');
  var ltF = $('legFThigh'), lsF = $('legFShin'), ltN = $('legNThigh'), lsN = $('legNShin');

  // 两段式腿部 IK：髋 -> 膝 -> 踏板
  function solve(hx, hy, px, py) {
    var dx = px - hx, dy = py - hy;
    var d = Math.hypot(dx, dy);
    var max = L1 + L2 - 0.5;
    if (d > max) { dx *= max / d; dy *= max / d; d = max; }
    var a = (L1 * L1 - L2 * L2 + d * d) / (2 * d);
    var h = Math.sqrt(Math.max(0, L1 * L1 - a * a));
    var ux = dx / d, uy = dy / d;
    var kx = hx + a * ux + h * uy, ky = hy + a * uy - h * ux;
    var kx2 = hx + a * ux - h * uy, ky2 = hy + a * uy + h * ux;
    if (kx2 > kx) { kx = kx2; ky = ky2; }
    var t1 = Math.atan2(ky - hy, kx - hx);
    var t2 = Math.atan2(py - ky, px - kx);
    return [t1 * 180 / Math.PI, (t2 - t1) * 180 / Math.PI];
  }

  var start = null;
  function frame(now) {
    if (start === null) start = now;
    var t = (((now - start) / 1000) % PERIOD) / PERIOD * Math.PI * 2;
    var deg = t * 180 / Math.PI;

    wheelR.setAttribute('transform', 'rotate(' + (deg * 1.6) + ' 255 385)');
    wheelF.setAttribute('transform', 'rotate(' + (deg * 1.6) + ' 525 385)');
    crankF.setAttribute('transform', 'rotate(' + deg + ' ' + BX + ' ' + BY + ')');
    crankN.setAttribute('transform', 'rotate(' + deg + ' ' + BX + ' ' + BY + ')');

    // 身体随蹬踏轻微起伏
    var bob = 2.4 * Math.sin(2 * t);
    var phi = 1.4 * Math.sin(2 * t) * Math.PI / 180;
    bodyG.setAttribute('transform', 'translate(0 ' + bob + ') rotate(' + (phi * 180 / Math.PI) + ' ' + BC.x + ' ' + BC.y + ')');

    // 髋部跟随身体的起伏与旋转
    function hipOf(h) {
      var dx = h.x - BC.x, dy = h.y - BC.y;
      return {
        x: BC.x + dx * Math.cos(phi) - dy * Math.sin(phi),
        y: BC.y + bob + dx * Math.sin(phi) + dy * Math.cos(phi)
      };
    }
    var pn = { x: BX + R * Math.cos(t), y: BY + R * Math.sin(t) };
    var pf = { x: BX - R * Math.cos(t), y: BY - R * Math.sin(t) };
    var hn = hipOf(HIP_N), hf = hipOf(HIP_F);
    var an = solve(hn.x, hn.y, pn.x, pn.y);
    var af = solve(hf.x, hf.y, pf.x, pf.y);

    ltN.setAttribute('transform', 'translate(' + hn.x + ' ' + hn.y + ') rotate(' + an[0] + ')');
    lsN.setAttribute('transform', 'translate(' + L1 + ' 0) rotate(' + an[1] + ')');
    ltF.setAttribute('transform', 'translate(' + hf.x + ' ' + hf.y + ') rotate(' + af[0] + ')');
    lsF.setAttribute('transform', 'translate(' + L1 + ' 0) rotate(' + af[1] + ')');

    // 喉囊轻弹
    var s = 1 + 0.05 * Math.sin(2 * t + 0.7);
    pouch.setAttribute('transform', 'translate(468 229) scale(1 ' + s + ') translate(-468 -229)');

    // 围巾飘动
    scarf.setAttribute('transform', 'rotate(' + (4 * Math.sin(6 * t) + 3 * Math.sin(9.3 * t)) + ' 374 244)');

    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();
</script>
