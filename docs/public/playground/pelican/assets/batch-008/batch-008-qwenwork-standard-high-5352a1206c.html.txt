    if (attrs) set(el, attrs);
    return el;

  function buildSpokes(g, c) {

      var a = i * Math.PI / 9, R = GEO.wheelR * 0.8;
      g.appendChild(mk('line', {
        x1: c.x - Math.cos(a) * R, y1: c.y - Math.sin(a) * R,
        x2: c.x + Math.cos(a) * R, y2: c.y + Math.sin(a) * R,
        stroke: '#aab6c2', 'stroke-width': 2.1
      }));
    }
  }

  }

  function smooth(pts) {
    var d = 'M ' + pts[0].x.toFixed(1) + ' ' + pts[0].y.toFixed(1);
    for (var i = 1; i < pts.length - 1; i++) {
      var mx = (pts[i].x + pts[i + 1].x) / 2, my = (pts[i].y + pts[i + 1].y) / 2;
      d += ' Q ' + pts[i].x.toFixed(1) + ' ' + pts[i].y.toFixed(1) + ' ' + mx.toFixed(1) + ' ' + my.toFixed(1);
    }
    var p = pts[pts.length - 1];
    return d + ' L ' + p.x.toFixed(1) + ' ' + p.y.toFixed(1);
  }

  // 两骨 IK：求膝关节。dir 控制膝盖朝向
  function knee(hip, target, l1, l2, dir) {
    var dx = target.x - hip.x, dy = target.y - hip.y;
    var d = Math.sqrt(dx * dx + dy * dy);
    var max = (l1 + l2) * 0.995, min = Math.abs(l1 - l2) + 1;
    if (d > max) { // 直接沿方向拉满
      return { x: hip.x + dx / d * l1, y: hip.y + dy / d * l1, stretched: true };
    }
    if (d < min) d = min;
    var a = Math.atan2(dy, dx);
    var c = (l1 * l1 + d * d - l2 * l2) / (2 * l1 * d);
    c = Math.max(-1, Math.min(1, c));
    var b = Math.acos(c);
    var ang = a + dir * b;
    return { x: hip.x + l1 * Math.cos(ang), y: hip.y + l1 * Math.sin(ang), stretched: false };
  }

  function drawLeg(ids, hip, pedal, dir) {
    var ankle = { x: pedal.x - 2, y: pedal.y - 9 };
    var k = knee(hip, ankle, GEO.femur, GEO.tibia, dir);
    set($(ids.thigh), { d: 'M ' + hip.x.toFixed(1) + ' ' + hip.y.toFixed(1) + ' L ' + k.x.toFixed(1) + ' ' + k.y.toFixed(1) });
    set($(ids.shin), { d: 'M ' + k.x.toFixed(1) + ' ' + k.y.toFixed(1) + ' L ' + ankle.x.toFixed(1) + ' ' + ankle.y.toFixed(1) });
    set($(ids.foot), { d: footPath(ankle.x, ankle.y, k) });
    return k;
  }

  function footPath(x, y, knee) {
    // 脚掌随踩踏角度轻微俯仰
    var tilt = Math.max(-3, Math.min(3, (x - GEO.bb.x) * 0.12));
    return 'M ' + (x - 11) + ' ' + (y - 5) +
           ' L ' + (x + 13) + ' ' + (y - 6 + tilt).toFixed(1) +
           ' Q ' + (x + 22) + ' ' + (y + 1) + ' ' + (x + 13) + ' ' + (y + 4) +
           ' L ' + (x - 10) + ' ' + (y + 5) + ' Z' +
           ' M ' + (x + 2) + ' ' + (y - 5.5).toFixed(1) + ' L ' + (x + 3) + ' ' + (y + 3.5).toFixed(1);
  }

  // 翅膀：肩（随躯干起伏）到把手（固定）的锥形翼
  function wingPath(shoulder, grip, droop, w1, w2) {
    var dx = grip.x - shoulder.x, dy = grip.y - shoulder.y;
    var L = Math.sqrt(dx * dx + dy * dy) || 1;
    var nx = -dy / L, ny = dx / L;
    var mx = (shoulder.x + grip.x) / 2, my = (shoulder.y + grip.y) / 2;
    var up1 = { x: shoulder.x + nx * w1, y: shoulder.y + ny * w1 };
    var up2 = { x: grip.x + nx * w2,     y: grip.y + ny * w2 };
    var lo2 = { x: grip.x - nx * w2 * 0.5, y: grip.y - ny * w2 * 0.5 };
    var lo1 = { x: shoulder.x - nx * w1,  y: shoulder.y - ny * w1 };
    var cu  = { x: mx + nx * (w1 + droop), y: my + ny * (w1 + droop) };
    var cl  = { x: mx - nx * (w1 + droop * 0.6), y: my - ny * (w1 + droop * 0.6) };
    return 'M ' + up1.x.toFixed(1) + ' ' + up1.y.toFixed(1) +
           ' Q ' + cu.x.toFixed(1) + ' ' + cu.y.toFixed(1) + ' ' + up2.x.toFixed(1) + ' ' + up2.y.toFixed(1) +
           ' L ' + lo2.x.toFixed(1) + ' ' + lo2.y.toFixed(1) +
           ' Q ' + cl.x.toFixed(1) + ' ' + cl.y.toFixed(1) + ' ' + lo1.x.toFixed(1) + ' ' + lo1.y.toFixed(1) + ' Z';
  }

  /* ---------- 主循环 ---------- */
  var last = 0;
  function frame(now) {
    if (!last) last = now;
    var dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    if (S.running) {
      var v = BASE * S.speed;
      S.dist += v * dt;
      S.wheel += (v / GEO.wheelR) * dt;          // 弧度
      S.crank += (v / GEO.wheelR) * S.cadence * dt;
      S.t += dt;
      S.dt = dt;
    } else {
      S.dt = 0;
    }
    render();
    requestAnimationFrame(frame);
  }

  function render() {
    var deg = S.wheel * 180 / Math.PI;

    /* 轮组 */
    set($('spokes-rear'),  { transform: 'rotate(' + deg.toFixed(2) + ' 300 370)' });
    set($('spokes-front'), { transform: 'rotate(' + deg.toFixed(2) + ' 600 370)' });
    set($('chainring'),    { transform: 'rotate(' + (S.crank * 180 / Math.PI).toFixed(2) + ' 435 368)' });
    set($('chain'),        { 'stroke-dashoffset': (-S.dist * 1.1).toFixed(1) });

    /* 路面与视差 */
    set($('centerline'), { 'stroke-dashoffset': (-(S.dist % 96)).toFixed(1) });
    set($('hills-far'),  { transform: 'translate(' + (-(S.dist * 0.09) % 480).toFixed(1) + ' 0)' });
    set($('hills-near'), { transform: 'translate(' + (-(S.dist * 0.42) % 340).toFixed(1) + ' 0)' });
    set($('pebbles'),    { transform: 'translate(' + (-(S.dist * 1.35) % 300).toFixed(1) + ' 0)' });
    clouds.forEach(function (c) {
      c.x -= (c.v + BASE * S.speed * 0.04) * S.dt;
      if (c.x < -150) c.x = 1120;
      set(c.el, { transform: 'translate(' + c.x.toFixed(1) + ' ' + c.y + ') scale(' + c.s + ')' });
    });

    /* 尘埃 */
    dust.forEach(function (d) {
      var p = (d.p + S.dist * 0.0022) % 1;
      var x = GEO.rear.x - 26 - p * 110;
      var y = 430 - Math.sin(p * Math.PI) * 16 - p * 6;
      var r = (1 - p) * 7 * S.speed + 1.2;
      set(d.el, {
        cx: x.toFixed(1), cy: y.toFixed(1), r: r.toFixed(1),
        opacity: (Math.sin(p * Math.PI) * 0.5 * Math.min(1, S.speed)).toFixed(2)
      });
    });

    /* 曲柄与踏板 */
    var a = S.crank;
    var pn = { x: GEO.bb.x + GEO.crankR * Math.cos(a), y: GEO.bb.y + GEO.crankR * Math.sin(a) };
    var pf = { x: GEO.bb.x - GEO.crankR * Math.cos(a), y: GEO.bb.y - GEO.crankR * Math.sin(a) };
    set($('crank-near'), { x1: GEO.bb.x, y1: GEO.bb.y, x2: pn.x.toFixed(1), y2: pn.y.toFixed(1) });
    set($('crank-far'),  { x1: GEO.bb.x, y1: GEO.bb.y, x2: pf.x.toFixed(1), y2: pf.y.toFixed(1) });
    set($('pedal-near'), { transform: 'translate(' + pn.x.toFixed(1) + ' ' + pn.y.toFixed(1) + ')' });
    set($('pedal-far'),  { transform: 'translate(' + pf.x.toFixed(1) + ' ' + pf.y.toFixed(1) + ')' });

    /* 躯干起伏 */
    var bob = Math.sin(a) * 3.1 + Math.sin(a * 2) * 0.9;
    var lean = Math.sin(a) * 0.7;
    set($('rider'), { transform: 'translate(0 ' + bob.toFixed(2) + ') rotate(' + lean.toFixed(2) + ' 380 240)' });
    set($('head'),  { transform: 'rotate(' + (-Math.sin(a * 2 + 0.5) * 1.8).toFixed(2) + ' 470 200)' });

    /* 喉囊抖动 */
    var ps = 1 + Math.sin(a * 2 + 1.1) * 0.09;
    set($('pouch'), { transform: 'translate(502 146) scale(1 ' + ps.toFixed(3) + ') translate(-502 -146)' });

    /* 瞳孔微微扫视 */
    set($('pupil'), {
      cx: (497 + Math.sin(S.t * 0.9) * 1.1).toFixed(2),
      cy: (135 + Math.sin(a) * 0.5).toFixed(2)
    });

    /* 双腿 IK（髋随躯干起伏） */
    var hipN = { x: GEO.hipNear.x, y: GEO.hipNear.y + bob };
    var hipF = { x: GEO.hipFar.x,  y: GEO.hipFar.y + bob };
    drawLeg({ thigh: 'thigh-near', shin: 'shin-near', foot: 'foot-near' }, hipN, pn, -1);
    drawLeg({ thigh: 'thigh-far',  shin: 'shin-far',  foot: 'foot-far'  }, hipF, pf, -1);

    /* 丝巾：从颈侧结扣向后飘，贴背部上缘波动 */
    var pts1 = [], pts2 = [];
    for (var i = 0; i < 8; i++) {
      var wave = Math.sin(S.t * 9 - i * 0.85) * (1 + i * 1.9);
      pts1.push({ x: 448 - i * 20, y: 184 + bob - i * 3.4 + wave });
      pts2.push({ x: 446 - i * 17, y: 189 + bob - i * 2.4 - wave * 0.62 });
    }
    set($('scarf1'), { d: smooth(pts1) });
    set($('scarf2'), { d: smooth(pts2) });

    /* 双翼握把 */
    set($('wing-near'), { d: wingPath({ x: GEO.shoulderNear.x, y: GEO.shoulderNear.y + bob }, GEO.grip, 9 + Math.sin(a) * 2, 21, 8) });
    set($('wing-far'),  {
      d: wingPath({ x: GEO.shoulderFar.x, y: GEO.shoulderFar.y + bob }, { x: GEO.grip.x - 12, y: GEO.grip.y + 6 }, 6, 18, 7),
      fill: '#c8c3b4'
    });
  }

  /* ---------- 控件 ---------- */
  var btnPlay = document.getElementById('btn-play');
  var btnReset = document.getElementById('btn-reset');
  var rngSpeed = document.getElementById('rng-speed');
  var outSpeed = document.getElementById('out-speed');
  var rngCad = document.getElementById('rng-cadence');
  var outCad = document.getElementById('out-cadence');

  function setRunning(on) {
    S.running = on;
    btnPlay.textContent = on ? '暂停' : '播放';
  }
  btnPlay.addEventListener('click', function () { setRunning(!S.running); });
  btnReset.addEventListener('click', function () {
    S.dist = 0; S.wheel = 0; S.crank = 0; S.t = 0; render();
  });
  rngSpeed.addEventListener('input', function () {
    S.speed = parseFloat(this.value);
    outSpeed.textContent = S.speed.toFixed(1) + '×';
  });
  rngCad.addEventListener('input', function () {
    S.cadence = parseFloat(this.value);
    outCad.textContent = S.cadence.toFixed(2);
  });
  document.addEventListener('keydown', function (e) {
    if (e.code === 'Space') { e.preventDefault(); setRunning(!S.running); }
    if (e.key === 'r' || e.key === 'R') { btnReset.click(); }
  });

  if (window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    setRunning(false);
  }

  render();
  requestAnimationFrame(frame);
})();
