  <script>
    (() => {
      const byId = (id) => document.getElementById(id);
      const elements = {
        rearWheel: byId('rearWheel'),
        frontWheel: byId('frontWheel'),
        cloudTrack: byId('cloudTrack'),
        waveTrack: byId('waveTrack'),
        roadsideTrack: byId('roadsideTrack'),
        roadTrack: byId('roadTrack'),
        body: byId('pelicanBody'),
        head: byId('pelicanHead'),
        eye: byId('pelicanEye'),
        scarf: byId('scarfTail'),
        chainring: byId('chainring'),
        crankArms: byId('crankArms'),
        backLeg: byId('backLeg'),
        backLegOutline: byId('backLegOutline'),
        frontLeg: byId('frontLeg'),
        frontLegOutline: byId('frontLegOutline'),
        backFoot: byId('backFoot'),
        frontFoot: byId('frontFoot'),
        backPedal: byId('backPedal'),
        frontPedal: byId('frontPedal'),
        playButton: byId('playButton'),
        playLabel: byId('playLabel'),
        pauseIcon: byId('pauseIcon'),
        playIcon: byId('playIcon'),
        speedRange: byId('speedRange'),
        speedValue: byId('speedValue')
      };

      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      const state = {
        playing: !reducedMotion.matches,
        speed: Number(elements.speedRange.value),
        distance: 0,
        time: 0,
        lastTime: null,
        frame: null
      };
      const wrap = (value, size) => ((value % size) + size) % size;

      function updateLeg(hip, ankle, prefix, bendDirection) {
        const upperLength = 126;
        const lowerLength = 126;
        const dx = ankle.x - hip.x;
        const dy = ankle.y - hip.y;
        const distance = Math.max(1, Math.min(Math.hypot(dx, dy), upperLength + lowerLength - 1));
        const ux = dx / distance;
        const uy = dy / distance;
        const along = (upperLength ** 2 - lowerLength ** 2 + distance ** 2) / (2 * distance);
        const height = Math.sqrt(Math.max(0, upperLength ** 2 - along ** 2));
        const knee = {
          x: hip.x + along * ux + bendDirection * height * -uy,
          y: hip.y + along * uy + bendDirection * height * ux
        };
        const path = `M${hip.x.toFixed(2)} ${hip.y.toFixed(2)}L${knee.x.toFixed(2)} ${knee.y.toFixed(2)}L${ankle.x.toFixed(2)} ${ankle.y.toFixed(2)}`;
        elements[`${prefix}Leg`].setAttribute('d', path);
        elements[`${prefix}LegOutline`].setAttribute('d', path);
        elements[`${prefix}Foot`].setAttribute('transform', `translate(${ankle.x.toFixed(2)} ${ankle.y.toFixed(2)})`);
      }

      function render() {
        const distance = state.distance;
        const pedalAngle = distance * 0.032 + 0.35;
        const wheelAngle = distance / 100 * 180 / Math.PI;
        const bob = Math.sin(pedalAngle * 2) * 3;
        const pedalX = Math.cos(pedalAngle) * 45;
        const pedalY = Math.sin(pedalAngle) * 45;
        const frontPedal = { x: 580 + pedalX, y: 560 + pedalY };
        const backPedal = { x: 580 - pedalX, y: 560 - pedalY };
        const frontAnkle = { x: frontPedal.x, y: frontPedal.y - 10 };
        const backAnkle = { x: backPedal.x, y: backPedal.y - 10 };

        elements.rearWheel.setAttribute('transform', `rotate(${wheelAngle.toFixed(3)})`);
        elements.frontWheel.setAttribute('transform', `rotate(${wheelAngle.toFixed(3)})`);
        elements.cloudTrack.setAttribute('transform', `translate(${-wrap(distance * 0.025, 1200)} 0)`);
        elements.waveTrack.setAttribute('transform', `translate(${-wrap(distance * 0.08, 1200)} 0)`);
        elements.roadsideTrack.setAttribute('transform', `translate(${-wrap(distance * 0.55, 1200)} 0)`);
        elements.roadTrack.setAttribute('transform', `translate(${-wrap(distance * 1.15, 1200)} 0)`);
        elements.body.setAttribute('transform', `translate(0 ${bob.toFixed(2)})`);
        elements.head.setAttribute('transform', `rotate(${(Math.sin(pedalAngle * 2 + 0.5) * 1.6).toFixed(2)} 590 185)`);
        elements.chainring.setAttribute('transform', `rotate(${(pedalAngle * 180 / Math.PI).toFixed(2)})`);
        elements.crankArms.setAttribute('d', `M580 560L${backPedal.x.toFixed(2)} ${backPedal.y.toFixed(2)}M580 560L${frontPedal.x.toFixed(2)} ${frontPedal.y.toFixed(2)}`);
        elements.frontPedal.setAttribute('transform', `translate(${frontPedal.x.toFixed(2)} ${frontPedal.y.toFixed(2)})`);
        elements.backPedal.setAttribute('transform', `translate(${backPedal.x.toFixed(2)} ${backPedal.y.toFixed(2)})`);

        updateLeg({ x: 492, y: 373 + bob }, backAnkle, 'back', -1);
        updateLeg({ x: 536, y: 374 + bob }, frontAnkle, 'front', -1);

        const scarfWave = Math.sin(state.time * 7) * 9;
        elements.scarf.setAttribute('d', `M554 222C510 ${204 + scarfWave * 0.25} 474 ${199 - scarfWave * 0.45} 431 ${211 + scarfWave}l15 21c40 ${-11 - scarfWave * 0.25} 71 ${4 + scarfWave * 0.2} 105 15Z`);

        const blinkPhase = state.time % 4.6;
        const eyeScale = blinkPhase > 4.42 ? Math.max(0.08, Math.abs((blinkPhase - 4.51) / 0.09)) : 1;
        elements.eye.setAttribute('transform', `translate(646 125) scale(1 ${eyeScale.toFixed(2)})`);
      }

      function tick(timestamp) {
        if (!state.playing) {
          state.frame = null;
          state.lastTime = null;
          return;
        }
        if (state.lastTime === null) state.lastTime = timestamp;
        const delta = Math.min((timestamp - state.lastTime) / 1000, 0.05);
        state.lastTime = timestamp;
        state.time += delta;
        state.distance += delta * state.speed;
        render();
        state.frame = requestAnimationFrame(tick);
      }

      function updatePlayControl() {
        elements.playLabel.textContent = state.playing ? '暂停' : '播放';
        elements.playButton.setAttribute('aria-label', state.playing ? '暂停动画' : '播放动画');
        elements.pauseIcon.style.display = state.playing ? '' : 'none';
        elements.playIcon.style.display = state.playing ? 'none' : '';
      }

      function setPlaying(playing) {
        state.playing = playing;
        updatePlayControl();
        if (playing && state.frame === null) {
          state.lastTime = null;
          state.frame = requestAnimationFrame(tick);
        } else if (!playing && state.frame !== null) {
          cancelAnimationFrame(state.frame);
          state.frame = null;
          state.lastTime = null;
        }
      }

      elements.playButton.addEventListener('click', () => setPlaying(!state.playing));
      elements.speedRange.addEventListener('input', (event) => {
        state.speed = Number(event.target.value);
        elements.speedValue.value = `${Math.round(state.speed / 10)} km/h`;
      });
      document.addEventListener('keydown', (event) => {
        const tagName = event.target.tagName;
        if (event.code === 'Space' && tagName !== 'INPUT' && tagName !== 'BUTTON') {
          event.preventDefault();
          setPlaying(!state.playing);
        }
      });
      reducedMotion.addEventListener('change', (event) => {
        if (event.matches) setPlaying(false);
      });

      render();
      updatePlayControl();
      if (state.playing) state.frame = requestAnimationFrame(tick);
    })();
  </script>
