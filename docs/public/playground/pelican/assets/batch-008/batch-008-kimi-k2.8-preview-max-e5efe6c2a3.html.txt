<script>
(function () {
  var svg = document.getElementById('scene');
  var NS = 'http://www.w3.org/2000/svg';
  var TAU = Math.PI * 2;

  var BB = { x: 382, y: 388 };
  var CRANK = 40;
  var HIP_NEAR = { x: 346, y: 306 };
  var HIP_FAR  = { x: 326, y: 300 };
  var L1 = 80, L2 = 82;

  var speed = 190;
  var wheelA = 0, crankA = 0;

  function el(name, attrs, parent) {
    var n = document.createElementNS(NS, name);
    for (var k in attrs) n.setAttribute(k, attrs[k]);
    (parent || svg).appendChild(n);
    return n;
  }

  function buildWheel(gid, cx, cy) {
    var g = document.getElementById(gid);
    el('circle', { cx: cx, cy: cy, r: 62, fill: 'none', stroke: '#22223b', 'stroke-width': 9 }, g);
    el('circle', { cx: cx, cy: cy, r: 55, fill: 'none', stroke: '#c3ccd6', 'stroke-width': 2.5 }, g);
    var spokes = el('g', { 'class': 'spokes' }, g);
    for (var i = 0; i < 12; i++) {
      var a = i / 12 * TAU;
      el('line', {
        x1: cx, y1: cy,
        x2: cx + 54 * Math.cos(a), y2: cy + 54 * Math.sin(a),
        stroke: '#aeb9c4', 'stroke-width': 2
      }, spokes);
    }
    el('circle', { cx: cx, cy: cy, r: 7, fill: '#e63946', stroke: '#22223b', 'stroke-width': 3 }, g);
    return spokes;
  }

  var spokesRear = buildWheel('wheelRear', 272, 412);
  var spokesFront = buildWheel('wheelFront', 552, 412);

  var dashes = document.getElementById('dashes');
  var dashXs = [];
  for (var i = 0; i < 9; i++) dashXs.push(i * 130);

  var clouds = Array.prototype.map.call(document.querySelectorAll('.cloud'), function (c, i) {
    return { node: c, y: +c.dataset.y, v: +c.dataset.speed, x: 150 + i * 300 };
  });
  var lines = Array.prototype.map.call(document.querySelectorAll('.sl'), function (l, i) {
    l.setAttribute('y1', l.dataset.y); l.setAttribute('y2', l.dataset.y);
    return { node: l, y: +l.dataset.y, v: +l.dataset.speed, x: 200 + i * 320 };
  });

  var nearCrank = document.getElementById('nearCrank');
  var farCrank = document.getElementById('farCrank');
  var pelican = document.getElementById('pelican');
  var eye = document.getElementById('eye');

  var nearPedal = document.getElementById('nearPedal');
  var farPedal = document.getElementById('farPedal');
  el('rect', { x: -17, y: -5, width: 34, height: 10, rx: 4, fill: '#1a1b2e' }, nearPedal);
  el('rect', { x: -15, y: -4, width: 30, height: 8, rx: 4, fill: '#6c7790' }, farPedal);

  var nearFoot = document.getElementById('nearFoot');
  var farFoot = document.getElementById('farFoot');

  function solveKnee(h, f) {
    var dx = f.x - h.x, dy = f.y - h.y;
    var d = Math.sqrt(dx * dx + dy * dy);
    d = Math.max(Math.abs(L1 - L2) + 1, Math.min(L1 + L2 - 1, d));
    var ux = dx / d, uy = dy / d;
    var c = Math.max(-1, Math.min(1, (L1 * L1 + d * d - L2 * L2) / (2 * L1 * d)));
    var s = Math.sqrt(Math.max(0, 1 - c * c));
    var k = uy >= 0 ? 1 : -1;
    return {
      x: h.x + L1 * c * ux + k * L1 * s * uy,
      y: h.y + L1 * c * uy - k * L1 * s * ux
    };
  }

  function pedalPos(phase) {
    return {
      x: BB.x + CRANK * Math.sin(phase),
      y: BB.y - CRANK * Math.cos(phase)
    };
  }

  var last = performance.now();
  function frame(now) {
    var dt = Math.min(0.05, (now - last) / 1000);
    last = now;

    wheelA += speed / 62 * dt;
    crankA += 2.4 * dt;

    var deg = wheelA * 180 / Math.PI;
    spokesRear.setAttribute('transform', 'rotate(' + deg + ' 272 412)');
    spokesFront.setAttribute('transform', 'rotate(' + deg + ' 552 412)');

    nearCrank.setAttribute('transform', 'rotate(' + crankA * 180 / Math.PI + ' ' + BB.x + ' ' + BB.y + ')');
    farCrank.setAttribute('transform', 'rotate(' + (crankA * 180 / Math.PI + 180) + ' ' + BB.x + ' ' + BB.y + ')');

    var bob = 2.2 * Math.sin(2 * crankA);
    pelican.setAttribute('transform', 'translate(0 ' + bob.toFixed(2) + ')');

    var pN = pedalPos(crankA);
    var pF = pedalPos(crankA + Math.PI);
    var hN = { x: HIP_NEAR.x, y: HIP_NEAR.y + bob };
    var hF = { x: HIP_FAR.x, y: HIP_FAR.y + bob };
    var kN = solveKnee(hN, pN);
    var kF = solveKnee(hF, pF);

    document.getElementById('nearThigh').setAttribute('d', 'M' + hN.x + ' ' + hN.y + ' L' + kN.x + ' ' + kN.y);
    document.getElementById('nearShin').setAttribute('d', 'M' + kN.x + ' ' + kN.y + ' L' + pN.x + ' ' + pN.y);
    document.getElementById('farThigh').setAttribute('d', 'M' + hF.x + ' ' + hF.y + ' L' + kF.x + ' ' + kF.y);
    document.getElementById('farShin').setAttribute('d', 'M' + kF.x + ' ' + kF.y + ' L' + pF.x + ' ' + pF.y);

    nearFoot.innerHTML = '';
    farFoot.innerHTML = '';
    el('ellipse', { cx: pN.x, cy: pN.y - 3, rx: 13, ry: 6.5, fill: '#f4a261', transform: 'rotate(' + (-crankA * 180 / Math.PI) + ' ' + pN.x + ' ' + (pN.y - 3) + ')' }, nearFoot);
    el('ellipse', { cx: pF.x, cy: pF.y - 3, rx: 11, ry: 5.5, fill: '#d98e4a', transform: 'rotate(' + (-crankA * 180 / Math.PI) + ' ' + pF.x + ' ' + (pF.y - 3) + ')' }, farFoot);

    var pDeg = -crankA * 180 / Math.PI;
    nearPedal.setAttribute('transform', 'translate(' + pN.x + ' ' + pN.y + ') rotate(' + pDeg + ')');
    farPedal.setAttribute('transform', 'translate(' + pF.x + ' ' + pF.y + ') rotate(' + pDeg + ')');

    clouds.forEach(function (c) {
      c.x -= (c.v + speed * 0.12) * dt;
      if (c.x < -240) c.x = 960;
      c.node.setAttribute('transform', 'translate(' + c.x + ' ' + c.y + ')');
    });
    lines.forEach(function (l) {
      l.x -= (l.v + speed * 0.9) * dt;
      if (l.x < -160) l.x = 960;
      l.node.setAttribute('transform', 'translate(' + l.x + ' 0)');
    });
    dashXs.forEach(function (x, i) {
      x -= speed * dt;
      if (x < -130) x += 9 * 130;
      dashXs[i] = x;
      dashes.childNodes[i].setAttribute('transform', 'translate(' + x + ' 0)');
    });

    var phase = (now % 3400) / 3400;
    var s = 1;
    if (phase < 0.06) s = 1 - 0.92 * Math.sin(phase / 0.06 * Math.PI);
    eye.setAttribute('transform', 'translate(540 160) scale(1 ' + s.toFixed(3) + ') translate(-540 -160)');

    requestAnimationFrame(frame);
  }

  dashXs.forEach(function () { el('rect', { x: 0, y: 505, width: 64, height: 8, rx: 4 }, dashes); });
  requestAnimationFrame(frame);
})();
</script>
