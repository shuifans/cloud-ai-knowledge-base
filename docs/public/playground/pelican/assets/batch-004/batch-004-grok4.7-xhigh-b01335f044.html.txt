    function kneeOf(hip, foot, bend) {
      let dx = foot.x - hip.x;
      let dy = foot.y - hip.y;
      let dist = Math.hypot(dx, dy) || 0.001;
      const max = UPPER + LOWER - 1;
      const min = Math.abs(UPPER - LOWER) + 1;
      const clamped = Math.max(min, Math.min(max, dist));
      if (clamped !== dist) {
        const scale = clamped / dist;
        dx *= scale;
        dy *= scale;
        dist = clamped;
      }
      const base = Math.atan2(dy, dx);
      const cosA = (UPPER * UPPER + dist * dist - LOWER * LOWER) / (2 * UPPER * dist);
      const ang = Math.acos(Math.max(-1, Math.min(1, cosA)));
      const dir = base + bend * ang;
      return {
        x: hip.x + Math.cos(dir) * UPPER,
        y: hip.y + Math.sin(dir) * UPPER
      };
    }

    function stepDust(dt) {
      for (let i = 0; i < dust.length; i++) {

          p.el.setAttribute("opacity", 0);
          continue;

        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.vy += 30 * dt;
        const k = 1 - p.age / p.life;
        p.el.setAttribute("cx", p.x);
        p.el.setAttribute("cy", p.y);
        p.el.setAttribute("rx", 7 + p.age * 22);
