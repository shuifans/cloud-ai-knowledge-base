<script>
const $ = id => document.getElementById(id);
const NS = 'http://www.w3.org/2000/svg';

const BB = { x: 360, y: 385 };   // 中轴
const CRANK = 35;                // 曲柄长
const HIPX = 330, HIPY = 224;    // 髋关节
const L1 = 95, L2 = 108;         // 大腿 / 小腿长

// 辐条
for (const { g, cx, cy } of [
  { g: $('spokesR'), cx: 250, cy: 380 },
  { g: $('spokesF'), cx: 550, cy: 380 },
]) {
  for (let i = 0; i < 4; i++) {
    const a = Math.PI * i / 4 + Math.PI / 8;
    const l = document.createElementNS(NS, 'line');
    l.setAttribute('x1', (cx - 58 * Math.cos(a)).toFixed(1));
    l.setAttribute('y1', (cy - 58 * Math.sin(a)).toFixed(1));
    l.setAttribute('x2', (cx + 58 * Math.cos(a)).toFixed(1));
    l.setAttribute('y2', (cy + 58 * Math.sin(a)).toFixed(1));
    g.appendChild(l);
  }
}

// 云
const clouds = [];
for (const [x, y, s, v] of [[140, 86, 1, 13], [430, 52, 0.75, 20], [660, 120, 0.85, 9]]) {
  const g = document.createElementNS(NS, 'g');
  g.setAttribute('fill', '#fff');
  g.setAttribute('opacity', '0.9');
  for (const [ex, ey, rx, ry] of [[0, 0, 52, 20], [-30, 6, 30, 14], [28, 4, 34, 15]]) {
    const e = document.createElementNS(NS, 'ellipse');
    e.setAttribute('cx', ex * s);
    e.setAttribute('cy', ey * s);
    e.setAttribute('rx', rx * s);
    e.setAttribute('ry', ry * s);
    g.appendChild(e);
  }
  $('cloudLayer').appendChild(g);
  clouds.push({ g, x, y, v });
}

// 两段式腿的简易 IK：给定髋与脚踏位置求膝
function ik(hx, hy, fx, fy) {
  const dx = fx - hx, dy = fy - hy;
  let d = Math.hypot(dx, dy);
  d = Math.max(Math.abs(L1 - L2) + 0.5, Math.min(L1 + L2 - 0.5, d));
  const a = (L1 * L1 - L2 * L2 + d * d) / (2 * d);
  const h = Math.sqrt(Math.max(0, L1 * L1 - a * a));
  const ux = dx / d, uy = dy / d;
  return [hx + ux * a + uy * h, hy + uy * a - ux * h];
}

const deg = r => r * 180 / Math.PI;
const CRANK_W = 3.1, WHEEL_W = 8.5, ROAD_V = 290, CHAIN_V = 45;
let theta = 0, wheelA = 0, dash = 0, chT = 0, chB = 0;
let tPrev = performance.now();

function tick(now) {
  const dt = Math.min((now - tPrev) / 1000, 0.05);
  tPrev = now;
  theta += CRANK_W * dt;
  wheelA += WHEEL_W * dt;

  const bob = 2.4 * Math.sin(2 * theta);
  $('pelican').setAttribute('transform', `translate(0 ${bob.toFixed(2)})`);
  $('wing').setAttribute('transform', `rotate(${(-14 + 2.5 * Math.sin(2 * theta + 1)).toFixed(2)} 296 200)`);
  $('spokesR').setAttribute('transform', `rotate(${deg(wheelA).toFixed(1)} 250 380)`);
  $('spokesF').setAttribute('transform', `rotate(${deg(wheelA).toFixed(1)} 550 380)`);
  $('nearCrankG').setAttribute('transform', `rotate(${deg(theta).toFixed(1)} 360 385)`);
  $('farCrank').setAttribute('transform', `rotate(${deg(theta + Math.PI).toFixed(1)} 360 385)`);

  const hy = HIPY + bob;
  for (const [ph, legId, footId, pedalId] of [
    [theta, 'nearLeg', 'nearFoot', 'nearPedal'],
    [theta + Math.PI, 'farLeg', 'farFoot', 'farPedal'],
  ]) {
    const px = BB.x + CRANK * Math.cos(ph);
    const py = BB.y + CRANK * Math.sin(ph);
    const [kx, ky] = ik(HIPX, hy, px, py);
    $(legId).setAttribute('d', `M ${HIPX} ${hy.toFixed(1)} L ${kx.toFixed(1)} ${ky.toFixed(1)} L ${px.toFixed(1)} ${py.toFixed(1)}`);
    $(footId).setAttribute('cx', px.toFixed(1));
    $(footId).setAttribute('cy', (py - 2).toFixed(1));
    $(pedalId).setAttribute('x', (px - 9).toFixed(1));
    $(pedalId).setAttribute('y', (py - 3.5).toFixed(1));
  }

  dash = (dash + ROAD_V * dt) % 64;
  $('roadDash').setAttribute('stroke-dashoffset', dash.toFixed(1));
  chT = (chT - CHAIN_V * dt + 10) % 10;
  chB = (chB + CHAIN_V * dt) % 10;
  $('chainTop').setAttribute('stroke-dashoffset', chT.toFixed(2));
  $('chainBot').setAttribute('stroke-dashoffset', chB.toFixed(2));

  for (const c of clouds) {
    c.x -= c.v * dt;
    if (c.x < -140) c.x = 900 + Math.random() * 60;
    c.g.setAttribute('transform', `translate(${c.x.toFixed(1)} ${c.y})`);
  }

  requestAnimationFrame(tick);
}
requestAnimationFrame(tick);
</script>
