  <script>
    (() => {
      const $ = (id) => document.getElementById(id);
      const els = {
        rear: $('rear-wheel'), front: $('front-wheel'), farLeg: $('far-leg'), nearLeg: $('near-leg'),
        farKnee: $('far-knee'), nearKnee: $('near-knee'), farFoot: $('far-foot'), nearFoot: $('near-foot'),
        crankFar: $('crank-far'), crankNear: $('crank-near'), pedalFar: $('pedal-far'), pedalNear: $('pedal-near'),
        pelican: $('pelican'), wing: $('wing'), dashes: $('road-dashes'), plantsA: $('plants-a'), plantsB: $('plants-b'),
        cloudA: $('cloud-a'), cloudB: $('cloud-b'), cloudC: $('cloud-c'), wind: $('wind-lines'),
        button: $('toggle'), speed: $('speed'), speedValue: $('speed-value')
      };
      const x0 = 585, y0 = 500, radius = 35;
      let phase = 0, travel = 0, last = 0, playing = true;
      let speed = 1;

      function setLine(el, x1, y1, x2, y2) {
        el.setAttribute('x1', x1.toFixed(2)); el.setAttribute('y1', y1.toFixed(2));
        el.setAttribute('x2', x2.toFixed(2)); el.setAttribute('y2', y2.toFixed(2));
      }
      // Two-circle inverse kinematics keeps each knee and foot connected to its pedal.
      function leg(hipX, hipY, footX, footY, upper, lower) {
        const dx = footX - hipX, dy = footY - hipY;
        const d = Math.min(Math.hypot(dx, dy), upper + lower - .01);
        const along = (upper * upper - lower * lower + d * d) / (2 * d);
        const out = Math.sqrt(Math.max(0, upper * upper - along * along));
        const ux = dx / d, uy = dy / d;
        return { x: hipX + ux * along + uy * out, y: hipY + uy * along - ux * out };
      }
      function drawRider() {
        const bob = Math.sin(phase * 2) * 2.8;
        els.pelican.setAttribute('transform', `translate(0 ${bob.toFixed(2)})`);
        els.wing.setAttribute('transform', `rotate(${(Math.sin(phase + .7) * 2.3).toFixed(2)} 548 248)`);
        for (const side of ['far', 'near']) {
          const p = phase + (side === 'far' ? Math.PI : 0);
          const fx = x0 + radius * Math.cos(p), fy = y0 + radius * Math.sin(p);
          const hx = side === 'far' ? 542 : 567, hy = (side === 'far' ? 331 : 334) + bob;
          const knee = leg(hx, hy, fx, fy, 112, 111);
          els[side + 'Leg'].setAttribute('d', `M${hx.toFixed(2)} ${hy.toFixed(2)} Q${(hx + knee.x) / 2} ${(hy + knee.y) / 2 - 4} ${knee.x.toFixed(2)} ${knee.y.toFixed(2)} L${fx.toFixed(2)} ${(fy - 4).toFixed(2)}`);
          els[side + 'Knee'].setAttribute('cx', knee.x.toFixed(2));
          els[side + 'Knee'].setAttribute('cy', knee.y.toFixed(2));
          els[side + 'Foot'].setAttribute('transform', `translate(${fx.toFixed(2)} ${(fy - 7).toFixed(2)}) rotate(${(Math.sin(p) * 9).toFixed(2)})`);
          setLine(els[side === 'far' ? 'crankFar' : 'crankNear'], x0, y0, fx, fy);
          setLine(els[side === 'far' ? 'pedalFar' : 'pedalNear'], fx - 11, fy + 8, fx + 20, fy + 8);
        }
      }
      function wrapped(value, span) { return ((value % span) + span) % span; }
      function drawScene() {
        const wheelAngle = phase * 180 / Math.PI * 1.15;
        els.rear.setAttribute('transform', `translate(420 500) rotate(${wheelAngle.toFixed(2)})`);
        els.front.setAttribute('transform', `translate(770 500) rotate(${wheelAngle.toFixed(2)})`);
        els.dashes.setAttribute('transform', `translate(${-wrapped(travel * 1.7, 180).toFixed(2)} 0)`);
        const shrubX = -wrapped(travel * .65, 1220);
        els.plantsA.setAttribute('transform', `translate(${shrubX.toFixed(2)} 0)`);
        els.plantsB.setAttribute('transform', `translate(${shrubX.toFixed(2)} 0)`);
        els.cloudA.setAttribute('transform', `translate(${(155 - wrapped(travel * .1, 1510)).toFixed(2)} 190)`);
        els.cloudB.setAttribute('transform', `translate(${(642 - wrapped(travel * .07, 1600)).toFixed(2)} 244) scale(.72)`);
        els.cloudC.setAttribute('transform', `translate(${(1180 - wrapped(travel * .12, 1540)).toFixed(2)} 209) scale(.9)`);
        els.wind.setAttribute('transform', `translate(${(-wrapped(travel * 1.2, 38)).toFixed(2)} 0)`);
        drawRider();
      }
      function frame(now) {
        if (last) {
          const dt = Math.min((now - last) / 1000, .05);
          if (playing) { phase += dt * 3.15 * speed; travel += dt * 115 * speed; drawScene(); }
        }
        last = now;
        requestAnimationFrame(frame);
      }
      els.button.addEventListener('click', () => {
        playing = !playing;
        els.button.textContent = playing ? 'Ⅱ' : '▶';
        els.button.setAttribute('aria-label', playing ? '暂停动画' : '播放动画');
        els.button.title = playing ? '暂停动画' : '播放动画';
      });
      els.speed.addEventListener('input', () => {
        speed = Number(els.speed.value);
        els.speedValue.textContent = `${speed.toFixed(1)}×`;
      });
      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
        playing = false; els.button.textContent = '▶'; els.button.setAttribute('aria-label', '播放动画'); els.button.title = '播放动画';
      }
      drawScene();
      requestAnimationFrame(frame);
    })();
  </script>
