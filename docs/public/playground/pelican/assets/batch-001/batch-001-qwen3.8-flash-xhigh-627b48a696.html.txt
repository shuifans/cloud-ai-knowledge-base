<script>
(() => {
  const $ = id => document.getElementById(id);
  const TILE = 960;
  const BB = {x:410, y:418};          // 中轴（曲柄中心）
  const CRANK = 26;
  const HIP_NEAR = {x:374, y:302};
  const HIP_FAR  = {x:366, y:306};
  const L1 = 78, L2 = 76;              // 股骨 / 胫骨
  const R_WHEEL = 60;

  let running = true, speed = 9, dist = 0, t = 0, pop = 0, last = performance.now();

  // ---- 初始化程序化装饰 ----
  const stars = $('stars');
  for (let i = 0; i < 70; i++) {
    const c = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    c.setAttribute('cx', Math.random() * TILE);
    c.setAttribute('cy', 12 + Math.random() * 300);
    c.setAttribute('r', Math.random() * 1.5 + .4);
    c.setAttribute('opacity', .35 + Math.random() * .65);
    stars.appendChild(c);
  }
  // 为每个视差层补一份平铺副本
  document.querySelectorAll('#cloud-layer .tile,#hill-far-layer .tile,#hill-mid-layer .tile,#tree-layer .tile,#road-dash .tile')
    .forEach(tile => {
      const g = tile.parentNode;
      const c = tile.cloneNode(true);
      c.setAttribute('transform', 'translate(' + TILE + ',0)');
      c.removeAttribute('class');
      g.appendChild(c);
    });

  // ---- 两节骨骼 IK：已知髋、目标脚、两段骨长，求膝 ----
  function ik(hip, target, l1, l2, kneeForward) {
    let dx = target.x - hip.x, dy = target.y - hip.y;
    let d = Math.hypot(dx, dy) || 0.001;
    const dc = Math.min(Math.max(d, Math.abs(l1 - l2) + 1), l1 + l2 - 1);
    const ux = dx / d, uy = dy / d;
    const foot = {x: hip.x + ux * dc, y: hip.y + uy * dc};
    const base = Math.atan2(dy, dx);
    const a = Math.acos((l1 * l1 + dc * dc - l2 * l2) / (2 * l1 * dc));
    const ang = base + (kneeForward ? -a : a);
    const knee = {x: hip.x + l1 * Math.cos(ang), y: hip.y + l1 * Math.sin(ang)};
    return {knee, foot};
  }

  function setLeg(pfx, hip, pedal) {
    const {knee, foot} = ik(hip, pedal, L1, L2, true);
    const f = $(pfx + '-femur'), tibia = $(pfx + '-tibia'), footEl = $(pfx + '-foot');
    f.setAttribute('x1', hip.x); f.setAttribute('y1', hip.y);
    f.setAttribute('x2', knee.x); f.setAttribute('y2', knee.y);
    tibia.setAttribute('x1', knee.x); tibia.setAttribute('y1', knee.y);
    tibia.setAttribute('x2', foot.x); tibia.setAttribute('y2', foot.y);
    const dir = Math.atan2(foot.y - knee.y, foot.x - knee.x) * 180 / Math.PI;
    footEl.setAttribute('transform', `translate(${foot.x},${foot.y}) rotate(${(dir - 90) * .45})`);
    return {knee, foot};
  }

  function pedalPoint(angleDeg) {
    const r = angleDeg * Math.PI / 180;
    return {x: BB.x + CRANK * Math.cos(r), y: BB.y + CRANK * Math.sin(r)};
  }

  function offset(px, el, factor) {
    const x = (px * factor) % TILE;
    el.setAttribute('transform', 'translate(' + (-x) + ',0)');
  }

  function frame(now) {
    const dt = Math.min((now - last) / 1000, .05); last = now;
    if (running) { t += dt; dist += speed * 13 * dt; pop = Math.max(0, pop - dt); }

    const wheelDeg = dist / R_WHEEL * 180 / Math.PI;
    const crankDeg = wheelDeg * .45;                       // 齿比
    const a1 = crankDeg, a2 = crankDeg + 180;

    $('spin-rear').setAttribute('transform', `rotate(${wheelDeg})`);
    $('spin-front').setAttribute('transform', `rotate(${wheelDeg})`);
    $('chainring-spokes').setAttribute('transform', `rotate(${crankDeg} ${BB.x} ${BB.y})`);
    $('chain').setAttribute('stroke-dashoffset', -dist * .16);

    // 曲柄（绕中轴旋转，pedal 在局部 x=26 处）
    $('crank-near').setAttribute('transform', `translate(${BB.x},${BB.y}) rotate(${a1})`);
    $('crank-far').setAttribute('transform', `translate(${BB.x},${BB.y}) rotate(${a2})`);

    // 腿：脚被踏板拉着走
    const legPhase = crankDeg * Math.PI / 180;
    const nearPedal = pedalPoint(a1), farPedal = pedalPoint(a2);
    // 让脚在踏板上下方略有吸附，视觉上更贴
    setLeg('near', HIP_NEAR, {x: nearPedal.x, y: nearPedal.y - 6});
    setLeg('far',  HIP_FAR,  {x: farPedal.x,  y: farPedal.y - 6});

    // 车身起伏：踏频 2 次/圈 + 起飞
    const bob = Math.sin(legPhase * 2) * 3 + Math.sin(t * 1.3) * 1.2;
    const lift = pop > 0 ? -Math.sin((1 - pop / .9) * Math.PI) * 46 : 0;
    const lean = pop > 0 ? -Math.sin((1 - pop / .9) * Math.PI) * 7 : 0;
    $('bike').setAttribute('transform', `translate(0,${bob + lift}) rotate(${lean} 430 430)`);
    $('shadow').setAttribute('rx', 180 + lift * 0.7);
    $('shadow').setAttribute('opacity', .34 + lift * .004);

    // 翅膀拍打 + 头颈点动 + 喉囊抖动
    const flap = Math.sin(t * 5.2) * 13 - 4;
    $('wing').setAttribute('transform', `rotate(${flap} 428 250)`);
    $('head-neck').setAttribute('transform',
      `rotate(${Math.sin(t * 2.6) * 3.2 - 1} 452 252)`);
    const wob = Math.sin(t * 7.1);
    $('pouch').setAttribute('transform',
      `translate(0,${wob * 2}) translate(0,199) scale(1,${1 + wob * .05}) translate(0,-199)`);
    $('fish').setAttribute('transform',
      `translate(${604 + Math.sin(t * 3) * 3},${232 + wob * 2}) rotate(${Math.sin(t * 4) * 12})`);

    // 视差
    offset(dist, $('cloud-layer'), .05);
    offset(dist, $('hill-far-layer'), .18);
    offset(dist, $('hill-mid-layer'), .42);
    offset(dist, $('tree-layer'), .95);
    offset(dist, $('road-dash'), 1);
    $('sun-rays').setAttribute('transform', `rotate(${t * 6} 140 96)`);

    // 速度线
    $('speed-lines').setAttribute('opacity', Math.max(0, Math.min(.5, (speed - 6) / 34)));
    $('speed-lines').setAttribute('transform', `translate(${-((dist * 1.6) % 120)},0)`);

    $('kmh').textContent = speed.toFixed(1);
    $('odo').textContent = Math.round(dist * .06);
    requestAnimationFrame(frame);
  }

  $('btn-play').onclick = e => {
    running = !running;
    e.target.textContent = running ? '⏸ 暂停' : '▶ 播放';
  };
  $('speed').oninput = e => speed = +e.target.value;
  $('btn-night').onclick = e => {
    const on = document.body.classList.toggle('night');
    e.target.textContent = on ? '☀️ 白天' : '🌙 夜景';
  };
  $('btn-pop').onclick = () => {
    if (pop > 0) return;
    pop = .9;
    if (!running) { running = true; $('btn-play').textContent = '⏸ 暂停'; }
  };

  if (matchMedia('(prefers-reduced-motion: reduce)').matches) {
    running = false; $('btn-play').textContent = '▶ 播放';
  }
  requestAnimationFrame(frame);
})();
</script>
