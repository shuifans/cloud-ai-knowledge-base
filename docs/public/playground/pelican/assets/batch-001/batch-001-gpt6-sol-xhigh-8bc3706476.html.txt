  <script>
    (() => {
      const $ = id => document.getElementById(id);
      const rearWheel = $('rear-wheel'), frontWheel = $('front-wheel');
      const crank = $('crank'), rider = $('rider'), clouds = $('clouds'), speedLines = $('speed-lines');
      const farLeg = $('far-leg'), nearLeg = $('near-leg');
      const farFoot = $('far-foot'), nearFoot = $('near-foot');
      const button = $('toggle');
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      let playing = !reducedMotion.matches;
      let elapsed = 0, lastFrame = 0, frameId = 0;
      const TAU = Math.PI * 2;
      const round = n => Number(n.toFixed(2));

      function knee(hipX, hipY, footX, footY, upper = 130, lower = 143) {
        const dx = footX - hipX, dy = footY - hipY;
        const distance = Math.min(Math.hypot(dx, dy), upper + lower - .01);
        const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const across = Math.sqrt(Math.max(0, upper * upper - along * along));
        return [hipX + along * dx / distance + across * dy / distance,
                hipY + along * dy / distance - across * dx / distance];
      }

      function drawLeg(path, foot, hipX, hipY, pedalX, pedalY) {
        const [kx, ky] = knee(hipX, hipY, pedalX, pedalY);
        path.setAttribute('d', `M${round(hipX)} ${round(hipY)}L${round(kx)} ${round(ky)}L${round(pedalX)} ${round(pedalY)}`);
        foot.setAttribute('transform', `translate(${round(pedalX)} ${round(pedalY)})`);
      }

      function draw() {
        const wheelAngle = (elapsed * 360 / .87) % 360;
        rearWheel.setAttribute('transform', `translate(316 590) rotate(${round(wheelAngle)})`);
        frontWheel.setAttribute('transform', `translate(860 590) rotate(${round(wheelAngle)})`);

        const crankAngle = elapsed * TAU / 1.46;
        const pedalAngle = crankAngle - Math.atan2(32, 52);
        const rightX = 568 + Math.hypot(52, 32) * Math.cos(pedalAngle);
        const rightY = 590 + Math.hypot(52, 32) * Math.sin(pedalAngle);
        const leftX = 1136 - rightX, leftY = 1180 - rightY;
        crank.setAttribute('transform', `rotate(${round(crankAngle * 180 / Math.PI)} 568 590)`);

        const bob = 2.5 * Math.sin(crankAngle * 2);
        rider.setAttribute('transform', `translate(0 ${round(bob)})`);
        drawLeg(farLeg, farFoot, 502, 390 + bob, rightX, rightY);
        drawLeg(nearLeg, nearFoot, 555, 390 + bob, leftX, leftY);
        clouds.setAttribute('transform', `translate(${round(10 * Math.sin(elapsed * .34))} 0)`);
        speedLines.setAttribute('opacity', round(.42 + .2 * Math.sin(elapsed * 8)));
      }

      function animate(now) {
        if (!playing) return;
        if (lastFrame) elapsed += Math.min((now - lastFrame) / 1000, .06);
        lastFrame = now;
        draw();
        frameId = requestAnimationFrame(animate);
      }

      function updateButton() {
        button.textContent = playing ? 'Ⅱ 暂停' : '▶ 继续';
        button.setAttribute('aria-label', playing ? '暂停动画' : '继续动画');
        button.setAttribute('aria-pressed', String(!playing));
      }

      button.addEventListener('click', () => {
        playing = !playing;
        updateButton();
        if (playing) { lastFrame = 0; frameId = requestAnimationFrame(animate); }
        else cancelAnimationFrame(frameId);
      });
      reducedMotion.addEventListener('change', event => {
        if (event.matches && playing) button.click();
      });
      draw();
      updateButton();
      if (playing) frameId = requestAnimationFrame(animate);
    })();
  </script>
