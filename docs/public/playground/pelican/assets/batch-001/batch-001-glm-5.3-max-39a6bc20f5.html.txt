<script>
(function(){
  'use strict';
  const NS = 'http://www.w3.org/2000/svg';
  const byId = id => document.getElementById(id);
  function el(tag, attrs, parent){
    const n = document.createElementNS(NS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    parent.appendChild(n);
    return n;
  }

  // 太阳光芒
  const rays = byId('rays');
  for (let k = 0; k < 12; k++){
    const a = k * 30 * Math.PI / 180;
    el('line', {
      x1: 110 + 46*Math.cos(a), y1: 90 + 46*Math.sin(a),
      x2: 110 + 64*Math.cos(a), y2: 90 + 64*Math.sin(a)
    }, rays);
  }

  // 辐条 + 气门嘴
  function buildSpokes(g, cx){
    for (let k = 0; k < 12; k++){
      const a = k * 30 * Math.PI / 180;
      el('line', {
        x1: cx + 10*Math.cos(a), y1: 346 + 10*Math.sin(a),
        x2: cx + 56*Math.cos(a), y2: 346 + 56*Math.sin(a)
      }, g);
    }
    el('rect', { x: cx - 2.5, y: 288, width: 5, height: 11, rx: 1.5, fill: '#565d66', stroke: 'none' }, g);
  }
  buildSpokes(byId('spokesR'), 280);
  buildSpokes(byId('spokesF'), 620);

  // 牙盘孔
  const ringG = byId('ringG');
  for (let k = 0; k < 5; k++){
    const a = (90 + 72*k) * Math.PI / 180;
    el('circle', { cx: 450 + 10*Math.cos(a), cy: 352 + 10*Math.sin(a), r: 3, fill: '#2b2f36', stroke: 'none' }, ringG);
  }

  // 路面虚线 / 草丛
  const dashes = byId('dashes');
  for (let i = -1; i <= 7; i++){
    el('rect', { x: i*140 + 20, y: 437, width: 62, height: 10, rx: 4, stroke: 'none' }, dashes);
  }
  function buildTufts(g, step, y){
    let d = '';
    for (let i = -1; i <= 6; i++){
      const x = i * step;
      d += `M ${x} ${y} l 4 -11 l 4 11 z M ${x+9} ${y} l 5 -14 l 5 14 z M ${x+19} ${y} l 4 -10 l 4 10 z `;
    }
    el('path', { d, stroke: 'none' }, g);
  }
  buildTufts(byId('tuftsA'), 173, 411);
  buildTufts(byId('tuftsB'), 224, 476);

  const T = {
    spokesR: byId('spokesR'), spokesF: byId('spokesF'), ringG: ringG,
    armN: byId('armN'), armF: byId('armF'),
    pedalN: byId('pedalN'), pedalF: byId('pedalF'),
    legN: byId('legN'), legF: byId('legF'),
    footN: byId('footN'), footF: byId('footF'),
    peli: byId('peli'), headG: byId('headG'), beakTop: byId('beakTop'), pouchG: byId('pouchG'),
    zoom: byId('zoom'), dashes: dashes,
    tuftsA: byId('tuftsA'), tuftsB: byId('tuftsB'),
    chainT: byId('chainT'), chainB: byId('chainB')
  };

  // 视差层：远山与云
  const layers = [
    { el: byId('hillA'),  x: 180, y: 0,   sp: .05, cyc: 1500, s: 1   },
    { el: byId('hillB'),  x: 640, y: 0,   sp: .10, cyc: 1700, s: 1   },
    { el: byId('cloud1'), x: 120, y: 70,  sp: .03, cyc: 1160, s: 1   },
    { el: byId('cloud2'), x: 470, y: 128, sp: .05, cyc: 1160, s: .8  },
    { el: byId('cloud3'), x: 800, y: 46,  sp: .04, cyc: 1160, s: .65 }
  ];

  const V = 460, R = 70, GEAR = 2.4, CR = 36, L1 = 95, L2 = 95;
  const BB = { x: 450, y: 352 };
  const hipN = { x: 398, y: 218 }, hipF = { x: 386, y: 212 };

  function ik(hx, hy, fx, fy){
    let dx = fx - hx, dy = fy - hy;
    let d = Math.hypot(dx, dy);
    const maxD = L1 + L2 - 0.5;
    if (d > maxD){
      dx *= maxD / d; dy *= maxD / d; d = maxD;
      fx = hx + dx; fy = hy + dy;
    }
    if (d < 1e-6) d = 1e-6;
    let c = (L1*L1 + d*d - L2*L2) / (2 * L1 * d);
    const a = Math.acos(Math.max(-1, Math.min(1, c)));
    const ang = Math.atan2(dy, dx) - a; // 膝盖朝前
    return { kx: hx + L1*Math.cos(ang), ky: hy + L1*Math.sin(ang), fx, fy };
  }

  let last = performance.now();
  let dist = 0, crank = -1.2, chainOff = 0;
  let speedMul = 1, targetMul = 1;
  const prm = matchMedia('(prefers-reduced-motion: reduce)').matches;

  const stage = byId('stage');
  stage.addEventListener('pointerdown', e => { e.preventDefault(); targetMul = 2.4; });
  ['pointerup','pointercancel','pointerleave'].forEach(ev =>
    stage.addEventListener(ev, () => { targetMul = 1; }));

  function setLeg(poly, hx, hy, fx, fy){
    const r = ik(hx, hy, fx, fy);
    poly.setAttribute('points',
      `${hx.toFixed(1)},${hy.toFixed(1)} ${r.kx.toFixed(1)},${r.ky.toFixed(1)} ${r.fx.toFixed(1)},${r.fy.toFixed(1)}`);
  }

  function frame(now){
    const dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    speedMul += (targetMul - speedMul) * Math.min(1, dt * 4);
    const v = V * speedMul;
    dist += v * dt;
    crank += v / R / GEAR * dt;

    const wDeg = dist / R * 180 / Math.PI;
    const cDeg = crank * 180 / Math.PI;
    T.spokesR.setAttribute('transform', `rotate(${wDeg.toFixed(1)} 280 346)`);
    T.spokesF.setAttribute('transform', `rotate(${wDeg.toFixed(1)} 620 346)`);
    T.ringG.setAttribute('transform', `rotate(${cDeg.toFixed(1)} 450 352)`);

    const p1 = { x: BB.x + CR*Math.cos(crank), y: BB.y + CR*Math.sin(crank) };
    const p2 = { x: BB.x - CR*Math.cos(crank), y: BB.y - CR*Math.sin(crank) };
    T.armN.setAttribute('x2', p1.x.toFixed(1)); T.armN.setAttribute('y2', p1.y.toFixed(1));
    T.armF.setAttribute('x2', p2.x.toFixed(1)); T.armF.setAttribute('y2', p2.y.toFixed(1));
    T.pedalN.setAttribute('transform', `translate(${p1.x.toFixed(1)} ${p1.y.toFixed(1)})`);
    T.pedalF.setAttribute('transform', `translate(${p2.x.toFixed(1)} ${p2.y.toFixed(1)})`);

    const bob = 2 * Math.sin(2 * crank);
    T.peli.setAttribute('transform', `translate(0 ${bob.toFixed(2)})`);

    const boost = Math.max(0, Math.min(1, (speedMul - 1) / 1.4));
    T.headG.setAttribute('transform', `rotate(${(-4*boost).toFixed(2)} 438 112)`);
    T.beakTop.setAttribute('transform', `rotate(${(-11*boost).toFixed(2)} 466 92)`);
    T.zoom.setAttribute('opacity', (boost * 0.55).toFixed(2));

    const s = 1 + 0.035 * Math.sin(2*crank + 1.2);
    T.pouchG.setAttribute('transform',
      `translate(468 106) scale(1 ${s.toFixed(4)}) translate(-468 -106)`);

    setLeg(T.legN, hipN.x, hipN.y + bob, p1.x, p1.y - 9);
    setLeg(T.legF, hipF.x, hipF.y + bob*0.85, p2.x, p2.y - 9);
    T.footN.setAttribute('transform',
      `translate(${p1.x.toFixed(1)} ${(p1.y-9).toFixed(1)}) rotate(${(10*Math.sin(crank+0.6)).toFixed(1)})`);
    T.footF.setAttribute('transform',
      `translate(${p2.x.toFixed(1)} ${(p2.y-9).toFixed(1)}) rotate(${(10*Math.sin(crank+Math.PI+0.6)).toFixed(1)})`);

    T.dashes.setAttribute('transform', `translate(${-(dist % 140).toFixed(1)} 0)`);
    T.tuftsA.setAttribute('transform', `translate(${-(dist % 173).toFixed(1)} 0)`);
    T.tuftsB.setAttribute('transform', `translate(${-(dist % 224).toFixed(1)} 0)`);

    for (const l of layers){
      l.x -= v * l.sp * dt;
      if (l.x < -320) l.x += l.cyc;
      l.el.setAttribute('transform', `translate(${l.x.toFixed(1)} ${l.y}) scale(${l.s})`);
    }

    chainOff = (chainOff - (v / R / GEAR) * 17 * dt) % 520;
    T.chainT.style.strokeDashoffset = chainOff.toFixed(1);
    T.chainB.style.strokeDashoffset = (-chainOff).toFixed(1);

    if (!prm) requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();
</script>
