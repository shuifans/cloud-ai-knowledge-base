<script>
(function () {
  var BB = { x: 400, y: 365 };
  var CRANK = 32;
  var HIP = { x: 396, y: 250 };
  var L1 = 76, L2 = 82;
  var PERIOD = 1100;              // ms per crank revolution, matches wheel spin

  var el = function (id) { return document.getElementById(id); };
  var spokesR = el('spokesR'), spokesF = el('spokesF');
  var ringArms = el('chainringArms'), riderUpper = el('riderUpper');
  var parts = {
    far:  { thigh: el('thighFar'),  shin: el('shinFar'),  foot: el('footFar'),  arm: el('armFar'),  pedal: el('pedalFar') },
    near: { thigh: el('thighNear'), shin: el('shinNear'), foot: el('footNear'), arm: el('armNear'), pedal: el('pedalNear') }
  };

  // two-bone IK: knee bows forward (+x, the direction the bird faces)
  function knee(hx, hy, px, py) {
    var dx = px - hx, dy = py - hy;
    var d = Math.sqrt(dx * dx + dy * dy);
    d = Math.max(Math.abs(L1 - L2) + 0.5, Math.min(L1 + L2 - 0.5, d));
    var base = Math.atan2(dy, dx);
    var cos = (L1 * L1 + d * d - L2 * L2) / (2 * L1 * d);
    var bend = Math.acos(Math.max(-1, Math.min(1, cos)));
    var a = base - bend;
    return { x: hx + L1 * Math.cos(a), y: hy + L1 * Math.sin(a) };
  }

  function setLeg(side, hipX, hipY, px, py) {
    var p = parts[side];
    var k = knee(hipX, hipY, px, py);
    var ay = py - 5;                              // ankle rests on top of the pedal
    p.thigh.setAttribute('x1', hipX); p.thigh.setAttribute('y1', hipY);
    p.thigh.setAttribute('x2', k.x);  p.thigh.setAttribute('y2', k.y);
    p.shin.setAttribute('x1', k.x);   p.shin.setAttribute('y1', k.y);
    p.shin.setAttribute('x2', px);    p.shin.setAttribute('y2', ay);
    p.foot.setAttribute('transform', 'translate(' + px + ',' + ay + ')');
    p.arm.setAttribute('x2', px);     p.arm.setAttribute('y2', py);
    p.pedal.setAttribute('transform', 'translate(' + px + ',' + py + ')');
  }

  function frame(now) {
    var t = (now % PERIOD) / PERIOD;
    var deg = t * 360;

    spokesR.setAttribute('transform', 'rotate(' + deg + ')');
    spokesF.setAttribute('transform', 'rotate(' + deg + ')');
    ringArms.setAttribute('transform', 'rotate(' + deg + ')');

    var bob = Math.sin(t * Math.PI * 4) * 1.7;    // two bobs per revolution
    riderUpper.setAttribute('transform', 'translate(0,' + bob + ')');

    var aNear = t * Math.PI * 2;
    var aFar = aNear + Math.PI;
    setLeg('near', HIP.x, HIP.y + bob, BB.x + CRANK * Math.cos(aNear), BB.y + CRANK * Math.sin(aNear));
    setLeg('far',  HIP.x, HIP.y + bob, BB.x + CRANK * Math.cos(aFar),  BB.y + CRANK * Math.sin(aFar));

    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();
</script>
