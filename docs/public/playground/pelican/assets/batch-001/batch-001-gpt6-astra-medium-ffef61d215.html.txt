<script>
const el=id=>document.getElementById(id);
let running=!matchMedia('(prefers-reduced-motion: reduce)').matches,phase=0,last=0;
const speed=el('speed');
function sync(){el('toggle').textContent=running?'暂停骑行':'继续骑行';el('toggle').setAttribute('aria-pressed',String(!running));}
el('toggle').addEventListener('click',()=>{running=!running;sync()});
speed.addEventListener('input',()=>{el('speed-value').textContent=Number(speed.value).toFixed(1)+'×'});
function leg(prefix,x,y,hipX,hipY,bend){
const dx=x-hipX,dy=y-hipY,dist=Math.hypot(dx,dy),length=128;
const offset=Math.sqrt(Math.max(0,length*length-dist*dist/4));
const kx=(hipX+x)/2+bend*dy/dist*offset,ky=(hipY+y)/2-bend*dx/dist*offset;
const d=`M${hipX} ${hipY} L${kx} ${ky} L${x} ${y-8}`;
el(prefix+'-leg').setAttribute('d',d);
if(prefix==='near')el('near-outline').setAttribute('d',d);
el(prefix+'-foot').setAttribute('d',`M${x-6} ${y-16} L${x-12} ${y-2} L${x+28} ${y-2} L${x+15} ${y-12} L${x+5} ${y-10} Z`);
}
function draw(){
const a=phase*Math.PI*2,bob=Math.sin(a*2)*2.2;
const x=449+49*Math.cos(a),y=526+49*Math.sin(a),fx=898-x,fy=1052-y;
for(const [id,cx] of [['rear-spokes',264],['front-spokes',664]])el(id).setAttribute('transform',`rotate(${phase*360*1.65} ${cx} 526)`);
el('cranks').setAttribute('d',`M${fx} ${fy} L${x} ${y}`);
el('pedals').setAttribute('d',`M${x-15} ${y}h35 M${fx-15} ${fy}h35`);
leg('far',fx,fy,398,325+bob,1);leg('near',x,y,391,334+bob,-1);
el('bird').setAttribute('transform',`translate(0 ${bob})`);
el('scarf').setAttribute('transform',`rotate(${Math.sin(a*2)*4} 460 205)`);
el('road').setAttribute('transform',`translate(${-((phase*180)%180)} 0)`);
el('clouds').setAttribute('transform',`translate(${ -((phase*7+250)%1200)+250} 0)`);
}
function frame(now){if(last&&running)phase+=Math.min((now-last)/1000,.05)*Number(speed.value)*.7;last=now;draw();requestAnimationFrame(frame)}
sync();draw();requestAnimationFrame(frame);
