  <script>
    (() => {
      const $ = id => document.getElementById(id);
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      const toggle = $('toggle');
      const speedInput = $('speed');
      const speedValue = $('speedValue');
      let playing = !reducedMotion.matches;
      let speed = 1;
      let phase = 0;
      let last = 0;

      function setPlayState() {
        toggle.innerHTML = playing ? '❚❚ &nbsp;暂停' : '▶ &nbsp;播放';
        toggle.setAttribute('aria-label', playing ? '暂停动画' : '播放动画');
      }
      toggle.addEventListener('click', () => { playing = !playing; setPlayState(); });
      speedInput.addEventListener('input', () => {
        speed = Number(speedInput.value);
        speedValue.value = `${speed}×`;
      });
      reducedMotion.addEventListener('change', event => { playing = !event.matches; setPlayState(); });

      const crankX = 578, crankY = 510, crankRadius = 40;
      const set = (id, attr, value) => $(id).setAttribute(attr, value);
      function leg(side, angle) {
        const front = side === 'front';
        const hipX = front ? 522 : 510;
        const hipY = front ? 365 : 360;
        const px = crankX + Math.cos(angle) * crankRadius;
        const py = crankY + Math.sin(angle) * crankRadius;
        const dx = px - hipX, dy = py - hipY;
        const distance = Math.hypot(dx, dy);
        const thigh = 105, shin = 110;
        const clamped = Math.min(distance, thigh + shin - .01);
        const along = (thigh * thigh - shin * shin + clamped * clamped) / (2 * clamped);
        const height = Math.sqrt(Math.max(0, thigh * thigh - along * along));
        const ux = dx / distance, uy = dy / distance;
        const kneeX = hipX + ux * along - uy * height;
        const kneeY = hipY + uy * along + ux * height;
        const prefix = front ? 'front' : 'back';
        set(`${prefix}LegPath`, 'd', `M${hipX} ${hipY}L${kneeX.toFixed(1)} ${kneeY.toFixed(1)}L${px.toFixed(1)} ${py.toFixed(1)}`);
        set(`${prefix}Foot`, 'd', `M${(px - 13).toFixed(1)} ${(py + 3).toFixed(1)}l27 0`);
        set(`${prefix}Crank`, 'd', `M${crankX} ${crankY}L${px.toFixed(1)} ${py.toFixed(1)}`);
        set(`${prefix}PedalBar`, 'd', `M${(px - 18).toFixed(1)} ${py.toFixed(1)}h36`);
      }
      function render() {
        const rotation = phase * 180 / Math.PI;
        set('rearWheel', 'transform', `translate(365 510) rotate(${rotation})`);
        set('frontWheel', 'transform', `translate(805 510) rotate(${rotation})`);
        leg('front', phase + Math.PI / 2);
        leg('back', phase - Math.PI / 2);
        set('bird', 'transform', `translate(0 ${(-1.7 + Math.sin(phase * 2) * 2).toFixed(2)})`);
        const farOffset = (phase * 18) % 1200;
        set('farOne', 'transform', `translate(${-farOffset} 0)`);
        set('farTwo', 'transform', `translate(${1200 - farOffset} 0)`);
        const nearOffset = (phase * 46) % 1200;
        set('nearOne', 'transform', `translate(${-nearOffset} 0)`);
        set('nearTwo', 'transform', `translate(${1200 - nearOffset} 0)`);
        const roadOffset = (phase * 92) % 300;
        set('stripes', 'transform', `translate(${-roadOffset} 0)`);
        set('clouds', 'transform', `translate(${Math.sin(phase * .09) * 12} 0)`);
      }
      function tick(now) {
        if (!last) last = now;
        const dt = Math.min((now - last) / 1000, .05);
        last = now;
        if (playing) { phase += dt * 2.5 * speed; render(); }
        requestAnimationFrame(tick);
      }
      setPlayState();
      render();
      requestAnimationFrame(tick);
    })();
  </script>
