      <div class="speed-control"><label class="speed-label" for="ride-speed">骑行速度<output id="speed-output" for="ride-speed">1.0×</output></label><input id="ride-speed" type="range" min="0.4" max="2" step="0.1" value="1" aria-label="骑行速度" aria-valuetext="1.0 倍速度"></div>
    </div>
    <footer class="footer"><span class="wish">风很轻，路还长。</span><span class="key-hint"><kbd>空格</kbd>暂停 / 继续</span></footer>
    <noscript><p>请启用 JavaScript 播放动画。当前仍可欣赏完整的 SVG 插画。</p></noscript>
  </main>
  <script>
    (() => {
      'use strict';
      const el = id => document.getElementById(id);
      const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
      const ui = {
        button: el('play-toggle'), label: el('play-label'), status: el('ride-status'),
        pause: el('pause-icon'), play: el('play-icon'), speed: el('ride-speed'), output: el('speed-output')
      };
      const art = {};
      ['rear-wheel','front-wheel','cranks','far-leg-path','near-leg-path','near-leg-highlight',
       'far-foot','far-pedal','near-pedal','pelican','pelican-head','eye','closed-eye',
       'scarf-tail','scarf-tail-back','wing','wing-feathers','clouds','gulls','sailboat',
       'sea-lines','shrubs','road-marks','road-specks','foreground-grass','breeze','ground-shadow']
        .forEach(id => { art[id] = el(id); });
      let playing = !reducedMotion.matches;
      let speed = 1;
      let phase = .65;
      let previousTime = null;
      let frame = null;
      const TAU = Math.PI * 2;
      const fixed = n => n.toFixed(3);

      // Solve two connected leg segments so both webbed feet stay on the pedals.
      function legPath(hipX, hipY, footX, footY, side) {
        const upper = 80, lower = 91;
        const dx = footX - hipX, dy = footY - hipY;
        const distance = Math.min(Math.hypot(dx,dy),upper + lower - .001);
        const along = (upper * upper - lower * lower + distance * distance) / (2 * distance);
        const out = Math.sqrt(Math.max(0, upper * upper - along * along));
        const nx = dx / Math.hypot(dx,dy), ny = dy / Math.hypot(dx,dy);
        const kneeX = hipX + along * nx + side * out * ny;
        const kneeY = hipY + along * ny - side * out * nx;
        return `M${fixed(hipX)} ${fixed(hipY)}L${fixed(kneeX)} ${fixed(kneeY)}L${fixed(footX)} ${fixed(footY - 3)}`;
      }

      function draw() {
        const angle = phase % TAU;
        const travel = phase * 77 * 1.05;
        const wheelDegrees = phase * 1.05 * 180 / Math.PI;
        const crankDegrees = angle * 180 / Math.PI;
        const bob = Math.sin(angle * 2) * 2.1;
        art['rear-wheel'].setAttribute('transform',`rotate(${fixed(wheelDegrees)})`);
        art['front-wheel'].setAttribute('transform',`rotate(${fixed(wheelDegrees + 13)})`);
        art.cranks.setAttribute('transform',`rotate(${fixed(crankDegrees)} 586 483)`);
        const px = 586 + Math.cos(angle) * 28;
        const py = 483 + Math.sin(angle) * 28;
        const fx = 1172 - px, fy = 966 - py;
        art['near-pedal'].setAttribute('transform',`translate(${fixed(px)} ${fixed(py)})`);
        art['far-pedal'].setAttribute('transform',`translate(${fixed(fx)} ${fixed(fy)})`);
        art['far-foot'].setAttribute('transform',`translate(${fixed(fx)} ${fixed(fy)})`);
        const nearPath = legPath(554,348 + bob,px,py,1);
        art['near-leg-path'].setAttribute('d',nearPath);
        art['near-leg-highlight'].setAttribute('d',nearPath);
        art['far-leg-path'].setAttribute('d',legPath(573,347 + bob,fx,fy,1));
        art.pelican.setAttribute('transform',`translate(0 ${fixed(bob)})`);
        art['pelican-head'].setAttribute('transform',`rotate(${fixed(Math.sin(angle) * 1.1)} 677 171)`);

        // The outer wing reaches a fixed handlebar while its shoulder follows the body.
        const tip = -bob;
        art.wing.setAttribute('d',`M518 273C551 256 581 270 612 285L682 ${fixed(294+tip)} 714 ${fixed(282+tip)}Q722 ${fixed(280+tip)} 722 ${fixed(287+tip)}L716 ${fixed(295+tip)}Q726 ${fixed(287+tip)} 729 ${fixed(292+tip)}C732 ${fixed(298+tip)} 719 ${fixed(309+tip)} 706 ${fixed(310+tip)}L660 ${fixed(317+tip)}C628 316 603 309 579 310 553 314 527 310 518 295`);
        const flutter = Math.sin(phase * 2.4) * 5;
        const flutterBack = Math.sin(phase * 2.4 - .8) * 4;
        art['scarf-tail'].setAttribute('d',`M644 222C613 225 600 ${205+flutter} 561 ${211+flutter}L570 ${218+flutter} 563 ${226+flutter}C600 ${217+flutter} 615 241 645 233Z`);
        art['scarf-tail-back'].setAttribute('d',`M642 225C615 214 603 ${225+flutterBack} 572 ${216+flutterBack}L564 ${228+flutterBack}C598 ${238+flutterBack} 618 225 642 235Z`);
        const blinking = phase % 15 > 14.45 && phase % 15 < 14.95;
        art.eye.setAttribute('visibility',blinking ? 'hidden' : 'visible');
        art['closed-eye'].setAttribute('visibility',blinking ? 'visible' : 'hidden');
        art.clouds.setAttribute('transform',`translate(${fixed(-(travel*.035 % 1400))} 0)`);
        art['sea-lines'].setAttribute('transform',`translate(${fixed(-(travel*.13 % 1200))} 0)`);
        art.shrubs.setAttribute('transform',`translate(${fixed(-(travel*.42 % 1400))} 0)`);
        art['road-marks'].setAttribute('transform',`translate(${fixed(-(travel % 160))} 0)`);
        art['foreground-grass'].setAttribute('transform',`translate(${fixed(-(travel*.92 % 1400))} 0)`);
        art['road-specks'].setAttribute('transform',`translate(${fixed(-(travel % 1200))} 0)`);
        art.gulls.setAttribute('transform',`translate(${fixed(Math.sin(phase*.13)*18)} ${fixed(Math.sin(phase*.7)*2)})`);
        art.sailboat.setAttribute('transform',`translate(1032 ${fixed(353 + Math.sin(phase*.55)*1.6)}) rotate(${fixed(Math.sin(phase*.4)*1.3)})`);
        art.breeze.setAttribute('opacity',fixed(.55 + Math.sin(phase*2)*.25));
      }

      function tick(now) {
        frame = null;
        if (!playing || document.hidden) { previousTime = null; return; }
        if (previousTime !== null) phase += Math.min((now - previousTime) / 1000,.05) * 3.1 * speed;
        previousTime = now;
        draw();
        frame = requestAnimationFrame(tick);
      }
      function startLoop() {
        if (playing && !document.hidden && frame === null) frame = requestAnimationFrame(tick);
      }
      function syncUI() {
        ui.button.setAttribute('aria-pressed',String(!playing));
        ui.button.setAttribute('aria-label',playing ? '暂停动画' : '播放动画');
        ui.label.textContent = playing ? '歇一会儿' : '继续出发';
        ui.status.textContent = playing ? '正在享受海风' : '停下来，看看海';
        ui.pause.setAttribute('visibility',playing ? 'visible' : 'hidden');
        ui.play.setAttribute('visibility',playing ? 'hidden' : 'visible');
      }
      function setPlaying(value) {
        playing = value;
        previousTime = null;
        if (frame !== null) cancelAnimationFrame(frame);
        frame = null;
        syncUI();
        startLoop();
      }
      ui.button.addEventListener('click',() => setPlaying(!playing));
      ui.speed.addEventListener('input',() => {
        speed = Number(ui.speed.value);
        ui.output.value = `${speed.toFixed(1)}×`;
        ui.speed.setAttribute('aria-valuetext',`${speed.toFixed(1)} 倍速度`);
      });
      document.addEventListener('keydown',event => {
        if (event.code !== 'Space' || event.repeat || event.altKey || event.ctrlKey || event.metaKey) return;
        const target = event.target;
        if (target.isContentEditable || target.closest('button,input,textarea,select,a')) return;
        event.preventDefault();
        setPlaying(!playing);
      });
      document.addEventListener('visibilitychange',() => {
        previousTime = null;
        if (document.hidden && frame !== null) { cancelAnimationFrame(frame); frame = null; }
        else startLoop();
      });
      reducedMotion.addEventListener('change',event => { if (event.matches) setPlaying(false); });
      draw();
      syncUI();
      startLoop();
    })();
  </script>
