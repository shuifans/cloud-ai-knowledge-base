<script>
(() => {
  'use strict';
  const $ = id => document.getElementById(id);
  const ns = 'http://www.w3.org/2000/svg';
  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)');
  let playing = !reduceMotion.matches, speed = 1, time = 0, last = null;
  const make = (tag, attrs, parent) => { const el = document.createElementNS(ns, tag); for(const [key,value] of Object.entries(attrs)) el.setAttribute(key,value); parent.appendChild(el); return el; };
  const marks = Array.from({length:12},(_,i) => make('path',{d:`M${i*120} ${555+(i%3)*9}h${18+(i%4)*9}`},$('road')));
  const plants = Array.from({length:7},(_,i) => make('use',{href:'#tuft'},$('plants')));
  for(const id of ['pedalBack','pedalFront']) make('path',{d:'M-12 0H12',stroke:'#354e46','stroke-width':6,'stroke-linecap':'round'},$(id));
  function updateButton(){ $('playIcon').textContent=playing?'Ⅱ':'▶'; $('playLabel').textContent=playing?'歇一会儿':'继续出发'; $('toggle').setAttribute('aria-label',playing?'暂停动画':'播放动画'); $('toggle').setAttribute('aria-pressed',String(!playing)); }
  $('toggle').addEventListener('click',()=>{playing=!playing;updateButton();});
  $('speed').addEventListener('input',event=>{speed=Number(event.target.value);$('speedValue').value=speed.toFixed(1)+'×';});
  reduceMotion.addEventListener('change',event=>{playing=!event.matches;updateButton();});
  function draw(t) {
    const angle=t*2.65, deg=angle*180/Math.PI;
    const bob=Math.sin(angle*2)*1.5;
    $('rider').setAttribute('transform',`translate(0 ${bob})`);
    // Wheel circumference and road travel share the same linear speed.
    const travel=t*2.65*25;
    $('rearSpokes').setAttribute('transform',`rotate(${travel/83*180/Math.PI})`);
    $('frontSpokes').setAttribute('transform',`rotate(${travel/83*180/Math.PI})`);
    $('crank').setAttribute('transform',`rotate(${deg} 586 458)`);
    ['Back','Front'].forEach((side,index)=>{
      const a=angle+(index===0?Math.PI:0), x=586+25*Math.cos(a),y=458+25*Math.sin(a);
      const hipX=568,hipY=340,dx=x-hipX,dy=y-hipY,d=Math.hypot(dx,dy);
      const upper=77,lower=77,along=(upper*upper-lower*lower+d*d)/(2*d),height=Math.sqrt(Math.max(0,upper*upper-along*along));
      const kneeX=hipX+dx*along/d+dy*height/d,kneeY=hipY+dy*along/d-dx*height/d;
      $('leg'+side).setAttribute('d',`M${hipX} ${hipY}L${kneeX} ${kneeY}L${x} ${y-7}`);
      $('foot'+side).setAttribute('d',`M${x-5} ${y-10}Q${x+6} ${y-13} ${x+11} ${y-6}L${x+26} ${y-1}L${x+5} ${y+1}L${x-10} ${y-1}Z`);
      $('pedal'+side).setAttribute('transform',`translate(${x} ${y+3})`);
    });
    marks.forEach((el,i)=>el.setAttribute('transform',`translate(${((i*120-travel)%1440+1440)%1440-120-i*120} 0)`));
    plants.forEach((el,i)=>{ const x=((i*219-travel*.85)%1533+1533)%1533-140;el.setAttribute('transform',`translate(${x} ${i%2?511:484}) scale(${i%2?.7:.9})`); });
    $('clouds').setAttribute('transform',`translate(${-Math.sin(t*.045)*28} 0)`);
    $('boat').setAttribute('transform',`translate(${986-Math.sin(t*.07)*22} ${291+Math.sin(t*1.4)*2})`);
    $('water').setAttribute('transform',`translate(${Math.sin(t*.6)*9} 0)`);
    $('gull1').setAttribute('transform',`translate(${Math.sin(t*.3)*9} ${Math.sin(t*1.2)*3})`);
    $('gull2').setAttribute('transform',`translate(${Math.sin(t*.3+.8)*8} ${Math.sin(t*1.2+.8)*3})`);
    const wave=Math.sin(angle*2)*4;
    $('scarf').setAttribute('d',`M619 223Q574 ${221+wave} 552 ${203+wave}Q535 ${193-wave} 516 ${203-wave}L529 ${216-wave}L514 ${225-wave}Q550 ${217+wave} 576 ${235+wave}L616 240Z`);
  }
  function frame(now){ if(last!==null && playing && !document.hidden)time+=Math.min((now-last)/1000,.05)*speed;last=now;draw(time);requestAnimationFrame(frame); }
  document.addEventListener('visibilitychange',()=>{last=null;});
  updateButton();draw(0);requestAnimationFrame(frame);
})();
