(() => {
  'use strict';
  const $ = id => document.getElementById(id);
  const NS = 'http://www.w3.org/2000/svg';
  function element(tag, attrs, parent) { const el = document.createElementNS(NS, tag); Object.entries(attrs).forEach(([k,v]) => el.setAttribute(k,v)); parent.appendChild(el); return el; }
  function wheel(id, cx) {
    const group = $(id);
    element('circle', {cx,cy:446,r:83,fill:'#f4f2dc','fill-opacity':'.2',stroke:'#315b5d','stroke-width':9}, group);
    element('circle', {cx,cy:446,r:76,fill:'none',stroke:'#faf1d4','stroke-width':3}, group);
    const spokes = element('g', {}, group);
    for(let i=0;i<12;i++) { const a=i*Math.PI/6; element('line',{x1:cx,y1:446,x2:cx+75*Math.cos(a),y2:446+75*Math.sin(a),stroke:'#74958b','stroke-width':1.8},spokes); }
    element('path',{d:`M${cx-13} 372h13`,stroke:'#e5ac63','stroke-width':4},spokes);
    element('circle',{cx,cy:446,r:8,fill:'#315b5d'},group);
    return spokes;
  }
  const rear=wheel('rearWheel',470), front=wheel('frontWheel',710);
  let running=!matchMedia('(prefers-reduced-motion: reduce)').matches, speed=1, time=0, previous=null;
  const number=n=>n.toFixed(2);
  function limb(pathId, footId, hip, pedal, bend) {
    const dx=pedal[0]-hip[0],dy=pedal[1]-hip[1],d=Math.hypot(dx,dy);
    const upper=63,lower=65, a=(upper*upper-lower*lower+d*d)/(2*d),h=Math.sqrt(Math.max(0,upper*upper-a*a));
    const knee=[hip[0]+a*dx/d+bend*h*dy/d,hip[1]+a*dy/d-bend*h*dx/d];
    $(pathId).setAttribute('d',`M${hip.map(number)}L${knee.map(number)}L${pedal.map(number)}`);
    $(footId).setAttribute('d',`M${number(pedal[0]-7)} ${number(pedal[1]-2)}Q${number(pedal[0]+4)} ${number(pedal[1]-1)} ${number(pedal[0]+13)} ${number(pedal[1]-2)}`);
  }
  const clouds=[...$('clouds').children],plants=[...$('backPlants').children],near=[...$('frontPlants').children],marks=[...$('roadmarks').children];
  const wrap=(x,min,max)=>((x-min)%(max-min)+(max-min))%(max-min)+min;
  function draw(t) {
    const angle=t*2.7, bob=Math.sin(angle*2)*1.8;
    $('bird').setAttribute('transform',`translate(0 ${number(bob)})`);
    rear.setAttribute('transform',`rotate(${t*135} 470 446)`);front.setAttribute('transform',`rotate(${t*135} 710 446)`);
    const a=[590+22*Math.cos(angle),421+22*Math.sin(angle)],b=[590-22*Math.cos(angle),421-22*Math.sin(angle)];
    limb('legBack','footBack',[567,330+bob],b,1);limb('legFront','footFront',[576,330+bob],a,1);
    $('crankArm').setAttribute('d',`M${a.map(number)}L${b.map(number)}`);
    $('pedalFront').setAttribute('d',`M${a[0]-10} ${a[1]+4}h22`);$('pedalBack').setAttribute('d',`M${b[0]-10} ${b[1]+4}h22`);
    marks.forEach((el,i)=>el.setAttribute('x',wrap(40+i*300-t*195,-100,1400)));
    plants.forEach((el,i)=>el.setAttribute('x',wrap([98,934,996][i]-t*95,-100,1400)));
    near.forEach((el,i)=>el.setAttribute('x',wrap([210,243,1090][i]-t*240,-100,1450)));
    clouds.forEach((el,i)=>el.setAttribute('x',wrap([140,437,1040][i]-t*9,-150,1400)));
    $('gulls').setAttribute('transform',`translate(${Math.sin(t*.3)*12} ${Math.sin(t*1.6)*3})`);
    $('wind').setAttribute('opacity',number(.4+Math.sin(t*3)*.2));
    $('shadow').setAttribute('rx',number(176+Math.sin(angle*2)*3));
  }
  function updateButton() { $('playIcon').textContent=running?'Ⅱ':'▶';$('playText').textContent=running?'暂停骑行':'继续骑行';$('toggle').setAttribute('aria-label',running?'暂停动画':'播放动画'); }
  $('toggle').addEventListener('click',()=>{running=!running;updateButton();});
  $('speed').addEventListener('input',e=>{speed=Number(e.target.value);$('speedValue').textContent=speed.toFixed(1)+'×';});
  document.addEventListener('visibilitychange',()=>{previous=null;});
  function tick(now) { if(previous!==null && running && !document.hidden)time+=Math.min((now-previous)/1000,.05)*speed;previous=now;draw(time);requestAnimationFrame(tick); }
  draw(0);updateButton();requestAnimationFrame(tick);
})();
