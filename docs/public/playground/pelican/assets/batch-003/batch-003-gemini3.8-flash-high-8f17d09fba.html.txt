    // --- Sound Synthesis (Web Audio API) ---
    let audioCtx = null;
    function getAudioContext() {
      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
      if (audioCtx.state === 'suspended') {

      }

    function playBellChime() {
      if (!state.soundEnabled) return;
      try {
        const ctx = getAudioContext();
        const now = ctx.currentTime;
        
        // Classic dual-chime bicycle bell: two strikes with frequencies 2093Hz and 2637Hz
        const strikes = [0, 0.16];
        strikes.forEach((startTime) => {
          [2093, 2637].forEach((freq) => {
            const osc = ctx.createOscillator();
            const gain = ctx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(freq, now + startTime);
            
            gain.gain.setValueAtTime(0.25, now + startTime);
            gain.gain.exponentialRampToValueAtTime(0.0005, now + startTime + 0.45);
            
            osc.connect(gain);
            gain.connect(ctx.destination);
            
            osc.start(now + startTime);
            osc.stop(now + startTime + 0.5);
          });
        });
      } catch (err) {
        console.warn('Audio not allowed yet:', err);
      }
    }

    function playHonkQuack() {
      if (!state.soundEnabled) return;
      try {
        const ctx = getAudioContext();
        const now = ctx.currentTime;

        // Comical avian quack/honk with formant filtering
        const osc = ctx.createOscillator();
        const filter = ctx.createBiquadFilter();
        const gain = ctx.createGain();

        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(340, now);
        osc.frequency.exponentialRampToValueAtTime(210, now + 0.28);

        filter.type = 'bandpass';
        filter.frequency.setValueAtTime(680, now);
        filter.Q.setValueAtTime(3.5, now);

        gain.gain.setValueAtTime(0.28, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.3);

        osc.connect(filter);
        filter.connect(gain);
        gain.connect(ctx.destination);

        osc.start(now);
        osc.stop(now + 0.32);
      } catch (err) {
        console.warn('Audio not allowed yet:', err);
      }
    }

        const x = Math.random() * 900;

      }
      speedStreaks.innerHTML = html;
    }

    // --- 2D Inverse Kinematics Engine for Pelican Legs ---
    // Solves 2-segment limb: Hip -> Knee (L1) -> Foot (L2)
    function solveLegIK(hipX, hipY, targetX, targetY, L1, L2) {
      const dx = targetX - hipX;
      const dy = targetY - hipY;
      let dist = Math.hypot(dx, dy);

      // Clamp distance within reachable envelope
      const maxReach = L1 + L2 - 0.5;
      const minReach = Math.abs(L1 - L2) + 0.5;
      dist = Math.max(minReach, Math.min(maxReach, dist));

      const baseAngle = Math.atan2(dy, dx);

      // Law of cosines: L2^2 = L1^2 + dist^2 - 2*L1*dist*cos(angleAtHip)
      const cosHip = (L1 * L1 + dist * dist - L2 * L2) / (2 * L1 * dist);
      const hipOffset = Math.acos(Math.max(-1, Math.min(1, cosHip)));

      // Knee projects forward toward the bike handlebars
      const thighAngle = baseAngle - hipOffset;

      const kneeX = hipX + L1 * Math.cos(thighAngle);
      const kneeY = hipY + L1 * Math.sin(thighAngle);

      return {
        kneeX,
        kneeY,
        footX: targetX,
        footY: targetY
      };
    }

    // --- Main Physics & Animation Loop ---
    let lastTime = performance.now();
    let dustTimer = 0;

    function animate(now) {
      if (!state.isRunning) {
        lastTime = now;
        requestAnimationFrame(animate);
        return;
      }

      const dt = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;

      const speed = state.speedMultiplier;

      // 1. Advance Angles & Distances
      // Pedal rotation: 1.0 speed corresponds to ~1.4 revolutions/sec
      const pedalAngularSpeed = 3.6 * speed;
      state.pedalAngle = (state.pedalAngle + pedalAngularSpeed * dt) % (Math.PI * 2);

      // Wheel rotation: radius = 54px. Circumference = 2 * PI * 54 = 339.3px
      // Speed in px/sec: pedalAngularSpeed * 28px (gear ratio ~2.4) => ~240px/sec at 1.0x
      const roadSpeedPx = 250 * speed;
      const wheelRotationDeg = (roadSpeedPx * dt / (54 * 2 * Math.PI)) * 360;
      state.wheelAngle = (state.wheelAngle + wheelRotationDeg) % 360;

      // 2. Parallax Translations
      // Road divider dash offset
      state.groundOffset = (state.groundOffset + roadSpeedPx * dt) % 110;
      laneDivider1.setAttribute('stroke-dashoffset', state.groundOffset.toFixed(1));

      // Midground shift (fence & palms)
      fencePosts.forEach(p => {
        p.x -= roadSpeedPx * 0.45 * dt;
        if (p.x < -120) p.x += 1210;
      });
      palmTrees.forEach(p => {
        p.x -= roadSpeedPx * 0.35 * dt;
        if (p.x < -150) p.x += 1250;
      });
      renderMidground();

      // Roadside grass shift
      grassClumps.forEach(g => {
        g.x -= roadSpeedPx * dt;
        if (g.x < -40) g.x += 1090;
      });
      renderRoadside();

      // Clouds drift
      clouds.forEach(c => {
        c.x -= (c.speed * 60 + roadSpeedPx * 0.08) * dt;
        if (c.x < -120) c.x += 1220;
      });
      renderClouds();

      // 3. Spoke Rotation
      spokesRear.setAttribute('transform', `rotate(${state.wheelAngle.toFixed(1)})`);
      spokesFront.setAttribute('transform', `rotate(${state.wheelAngle.toFixed(1)})`);

      // 4. Pelican Bobbing & Spring Dynamics
      // Hip on saddle at (428, 325)
      // Torso bobs vertically and pitches slightly with pedaling effort
      const bobY = Math.sin(state.pedalAngle * 2) * (3.0 * Math.min(speed, 2.0));
      const bobPitch = Math.sin(state.pedalAngle) * (1.8 * Math.min(speed, 2.0));
      pelicanBodyGroup.setAttribute('transform', `translate(0, ${bobY.toFixed(2)}) rotate(${bobPitch.toFixed(2)}, 428, 325)`);

      // Pouch spring bounce: responds to acceleration + periodic jiggle
      const pouchTarget = Math.sin(state.pedalAngle * 2 - 0.7) * (4.5 * Math.min(speed, 2.5));
      const springK = 22.0;
      const damping = 4.5;
      const pouchForce = -springK * (state.pouchWobble - pouchTarget) - damping * state.pouchVelocity;
      state.pouchVelocity += pouchForce * dt;
      state.pouchWobble += state.pouchVelocity * dt;

      // Update Pouch SVG Path dynamically
      const pouchMidX = 595 + state.pouchWobble * 0.4;
      const pouchMidY = 242 + state.pouchWobble;
      pelicanPouch.setAttribute('d', `M 525 168 C 518 205, 545 ${pouchMidY + 10}, ${pouchMidX.toFixed(1)} ${pouchMidY.toFixed(1)} C 635 235, 655 195, 665 174 Z`);

      // Cap crest fluttering
      const crestWiggle = Math.sin(now * 0.012 * speed) * (4.0 * Math.min(speed, 2.0));
      headCrest.setAttribute('transform', `rotate(${crestWiggle.toFixed(1)}, 495, 150)`);

      // 5. Crankset & Pedal Kinematics
      const crankCenterX = 465;
      const crankCenterY = 425;
      const crankRadius = 26;

      // Crank angle: near side = state.pedalAngle; far side = state.pedalAngle + PI
      const nearAngle = state.pedalAngle;
      const farAngle = state.pedalAngle + Math.PI;

      const nearPedalX = crankCenterX + crankRadius * Math.cos(nearAngle);
      const nearPedalY = crankCenterY + crankRadius * Math.sin(nearAngle);

      const farPedalX = crankCenterX + crankRadius * Math.cos(farAngle);
      const farPedalY = crankCenterY + crankRadius * Math.sin(farAngle);

      // Update crank lines
      crankNear.setAttribute('x2', nearPedalX.toFixed(1));
      crankNear.setAttribute('y2', nearPedalY.toFixed(1));
      crankNearBolt.setAttribute('cx', nearPedalX.toFixed(1));
      crankNearBolt.setAttribute('cy', nearPedalY.toFixed(1));

      crankFar.setAttribute('x2', farPedalX.toFixed(1));
      crankFar.setAttribute('y2', farPedalY.toFixed(1));

      // Keep pedal platforms level (with gentle rocking angle)
      const nearPedalTilt = Math.sin(nearAngle) * 8;
      const farPedalTilt = Math.sin(farAngle) * 8;

      pedalNear.setAttribute('transform', `translate(${nearPedalX.toFixed(1)}, ${nearPedalY.toFixed(1)}) rotate(${nearPedalTilt.toFixed(1)}) translate(-11, -4)`);
      pedalFar.setAttribute('transform', `translate(${farPedalX.toFixed(1)}, ${farPedalY.toFixed(1)}) rotate(${farPedalTilt.toFixed(1)}) translate(-10, -4)`);

      // 6. Inverse Kinematics for Legs
      // Pelvis/Hip anchors (bob with the body)
      const hipNearX = 428;
      const hipNearY = 325 + bobY;
      const hipFarX = 424;
      const hipFarY = 322 + bobY;

      // Leg segment lengths
      const thighLen = 64;
      const shinLen = 62;

      // --- Solve Near Leg ---
      const nearIK = solveLegIK(hipNearX, hipNearY, nearPedalX, nearPedalY, thighLen, shinLen);

      // Thigh polygon
      const thighNearPath = `M ${hipNearX} ${hipNearY - 8} Q ${(hipNearX + nearIK.kneeX)/2 + 6} ${(hipNearY + nearIK.kneeY)/2 - 5} ${nearIK.kneeX} ${nearIK.kneeY} L ${nearIK.kneeX - 4} ${nearIK.kneeY + 4} Q ${(hipNearX + nearIK.kneeX)/2 - 6} ${(hipNearY + nearIK.kneeY)/2 + 8} ${hipNearX - 6} ${hipNearY + 4} Z`;
      thighNear.setAttribute('d', thighNearPath);
      kneeNearJoint.setAttribute('cx', nearIK.kneeX.toFixed(1));
      kneeNearJoint.setAttribute('cy', nearIK.kneeY.toFixed(1));

      // Shin polygon
      const shinNearPath = `M ${nearIK.kneeX + 4} ${nearIK.kneeY} L ${nearPedalX + 4} ${nearPedalY - 2} L ${nearPedalX - 4} ${nearPedalY + 2} L ${nearIK.kneeX - 4} ${nearIK.kneeY} Z`;
      shinNear.setAttribute('d', shinNearPath);

      // Foot placement & rotation on pedal
      footNear.setAttribute('transform', `translate(${nearPedalX.toFixed(1)}, ${nearPedalY.toFixed(1)}) rotate(${nearPedalTilt.toFixed(1)})`);

      // --- Solve Far Leg ---
      const farIK = solveLegIK(hipFarX, hipFarY, farPedalX, farPedalY, thighLen, shinLen);

      const thighFarPath = `M ${hipFarX} ${hipFarY - 7} Q ${(hipFarX + farIK.kneeX)/2 + 5} ${(hipFarY + farIK.kneeY)/2 - 4} ${farIK.kneeX} ${farIK.kneeY} L ${farIK.kneeX - 3} ${farIK.kneeY + 3} Q ${(hipFarX + farIK.kneeX)/2 - 5} ${(hipFarY + farIK.kneeY)/2 + 7} ${hipFarX - 5} ${hipFarY + 3} Z`;
      thighFar.setAttribute('d', thighFarPath);
      kneeFarJoint.setAttribute('cx', farIK.kneeX.toFixed(1));
      kneeFarJoint.setAttribute('cy', farIK.kneeY.toFixed(1));

      const shinFarPath = `M ${farIK.kneeX + 3} ${farIK.kneeY} L ${farPedalX + 3} ${farPedalY - 2} L ${farPedalX - 3} ${farPedalY + 2} L ${farIK.kneeX - 3} ${farIK.kneeY} Z`;
      shinFar.setAttribute('d', shinFarPath);

      footFar.setAttribute('transform', `translate(${farPedalX.toFixed(1)}, ${farPedalY.toFixed(1)}) rotate(${farPedalTilt.toFixed(1)})`);

      // 7. Blinking Animation
      state.blinkTimer -= dt;
      if (state.blinkTimer <= 0) {
        if (!state.isBlinking) {
          state.isBlinking = true;
          eyelid.setAttribute('height', '18');
          setTimeout(() => {
            eyelid.setAttribute('height', '0');
            state.isBlinking = false;
            state.blinkTimer = 2.5 + Math.random() * 3.5;
          }, 150);
        }
      }

      // 8. Dust Puff Generation behind Rear Wheel
      dustTimer += dt;
      if (dustTimer > (0.18 / Math.max(speed, 0.5))) {
        dustTimer = 0;
        spawnDust(320, 480, speed);
      }
      updateDust(dt);

      requestAnimationFrame(animate);
    }
    requestAnimationFrame(animate);
